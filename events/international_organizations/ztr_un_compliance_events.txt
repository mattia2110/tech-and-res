namespace = ztr_un_compliance

# UN reform requests, fired by ztr_un_compliance_on_yearly_pulse_country.
# Accepting starts the enactment, refusing gives ztr_un_condemned_modifier.

# UNESCO: education reform for countries with no schools or terakoya
ztr_un_compliance.1 = {
	dlc = dlc_tech_res
	type = country_event
	placement = root
	duration = 3
	title = ztr_un_compliance.1.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					has_law = law_type:law_terakoya
				}
				desc = ztr_un_compliance.1.d_terakoya
			}
			triggered_desc = {
				trigger = {
					always = yes
				}
				desc = ztr_un_compliance.1.d
			}
		}
	}
	flavor = ztr_un_compliance.1.f
	event_image = {
		video = "generic_education"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/ztr_onu_member.dds"

	trigger = {
		ztr_un_unesco_education_request_valid = yes
	}

	cancellation_trigger = {
		NOT = {
			ztr_is_un_member = yes
		}
	}

	immediate = {
		set_variable = {
			name = ztr_un_unesco_request_cooldown
			years = 5
		}
	}

	# Accept
	option = {
		name = ztr_un_compliance.1.a
		default_option = yes
		ztr_un_unesco_start_education_reform = yes
		ai_chance = {
			base = 25
			modifier = {
				trigger = {
					has_strategy = ai_strategy_industrial_expansion
				}
				add = 40
			}
			modifier = {
				trigger = {
					ztr_un_unesco_reform_support > 0
				}
				add = 40
			}
			modifier = {
				trigger = {
					ig:ig_intelligentsia ?= {
						is_in_government = yes
					}
				}
				add = 15
			}
			modifier = {
				trigger = {
					has_imf_modifier = yes
				}
				add = 20
			}
			modifier = {
				trigger = {
					literacy_rate < 0.25
				}
				add = 10
			}
		}
	}

	# Refuse
	option = {
		name = ztr_un_compliance.1.b
		ztr_un_refuse_compliance_request = yes
		ai_chance = {
			base = 25
			modifier = {
				trigger = {
					ztr_un_unesco_reform_support < 0
				}
				add = 40
			}
			modifier = {
				trigger = {
					has_law_or_variant = law_type:law_isolationism
				}
				add = 30
			}
			modifier = {
				trigger = {
					country_rank >= rank_value:great_power
				}
				add = 15
			}
			modifier = {
				trigger = {
					has_modifier = ztr_icc_investigation_modifier
				}
				add = 20
			}
		}
	}
}

				add = 15
			}
			modifier = {
				trigger = {
					has_imf_modifier = yes
				}
