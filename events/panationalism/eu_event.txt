# ===================================================================================
# The European Project
# 1    EU founded (chair)                    2  invitation to EEC members
# 10   council vote opened (humans)          11 parliament reapportioned
# 20   directives menu                       21 treaties menu (chair)
# 25   vote passed                           26 vote failed
#      (25/26 go to every member, humans and AI: they carry the political backlash)
# 30   Euro adopted                          31 ECB established
# 32   ECB stance changed
# 40   the Union's political orientation shifted (decennial review)
# 60   confederal treaty conference (every member decides for itself)
# 70   individual invitation to federal union
# (41-42 ideological drift and 61 eurozone-gated market were removed)
# ===================================================================================

# EU foundation - the other EEC members decide whether to follow
ztr_eu_event.2 = {
	dlc = dlc_tech_res
	type = country_event
	placement = root

	duration = 1

	title = ztr_eu_event.2.t
	desc = ztr_eu_event.2.d
	flavor = ztr_eu_event.1.f

	event_image = {
		video = "unspecific_signed_contract"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/ztr_eu.dds"

	immediate = {
		random_country = {
			limit = {
				has_variable = ztr_eu_founder
			}
			save_scope_as = eu_founder
		}
	}

	# Join the Union
	option = {
		name = ztr_eu_event.2.a
		default_option = yes
		trigger = {
			is_not_isolationist_switzerland = yes
		}
		# If a Country joins the EU, all of its European subjects will become independent.
		# Overseas dependencies are kept: the EEC founders joined while holding colonial empires.
		custom_tooltip = {
			text = ztr_eu_join_country_loose_all_subjects_tt
			every_subject_or_below = {
				limit = { ztr_is_european_country = yes }
				make_independent = yes
			}
		}
		change_infamy = -5
		ztr_eu_member_accession = yes
		ai_chance = {
			base = 80
			modifier = {
				trigger = {
					NOT = {
						ztr_eu_ideology_compatible = yes
					}
				}
				add = -60
			}
			modifier = {
				trigger = {
					"relations(scope:eu_founder)" >= relations_threshold:amicable
				}
				add = 20
			}
		}
	}

	# Stay out: the EEC dissolves beneath our feet
	option = {
		name = ztr_eu_event.2.b
		hidden_effect = {
			if = {
				limit = {
					has_modifier = ztr_eec_member
				}
				remove_modifier = ztr_eec_member
			}
		}
		every_country = {
			limit = {
				has_modifier = ztr_eu_member
			}
			change_relations = {
				country = root
				value = -20
			}
		}
		ai_chance = {
			base = 20
			modifier = {
				trigger = {
					NOT = {
						ztr_eu_ideology_compatible = yes
					}
				}
				add = 60
			}
		}
	}
}

# Treaty menu (chair only)
ztr_eu_event.21 = {
	dlc = dlc_tech_res
	type = country_event
	placement = root
	duration = 1
	title = ztr_eu_event.21.t
	desc = ztr_eu_event.21.d
	event_image = {
		video = "unspecific_signed_contract"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/ztr_eu.dds"
	immediate = {
		# Count members and Euro adopters to gate the federation treaty
		ztr_eu_count_members_and_euro = yes
	}

	# Convene the confederal treaty conference: no Council vote is held, every
	# member state receives an individual summons (event 60) and answers alone
	option = {
		name = ztr_eu_event.21.a
		trigger = {
			is_not_isolationist_switzerland = yes
			global_var:ztr_eu_phase = 2
			has_technology_researched = e_government
			# The chair leads the Confederation bloc, so it must be in the eurozone
			custom_tooltip = {
				text = ztr_eu_single_market_euro_tt
				has_modifier = ztr_euro_member
			}
			# The three common policies must already bind the Union together
			custom_tooltip = {
				text = ztr_eu_confed_agriculture_tt
				has_global_variable = ztr_eu_dir_agriculture
				global_var:ztr_eu_dir_agriculture = 1
			}
			custom_tooltip = {
				text = ztr_eu_confed_schengen_tt
				has_global_variable = ztr_eu_dir_schengen
				global_var:ztr_eu_dir_schengen = 1
			}
			custom_tooltip = {
				text = ztr_eu_confed_cohesion_tt
				has_global_variable = ztr_eu_dir_cohesion
				global_var:ztr_eu_dir_cohesion = 1
			}
		}
		custom_tooltip = ztr_eu_establish_confederation_tt
		# Hidden: the tooltip walker must not descend into the bloc creation,
		# it would evaluate every member's accession while the bloc does not exist yet
		hidden_effect = {
			ztr_eu_establish_single_market = yes
		}
		remove_global_variable = ztr_eu_button_opened
		remove_global_variable = ztr_eu_tmp_member_count
		remove_global_variable = ztr_eu_tmp_euro_count
		ai_chance = {
			base = 0
		}
	}

	# Invite every other confederated member to form the European Federation
	option = {
		name = ztr_eu_event.21.b
		trigger = {
			is_not_isolationist_switzerland = yes
			global_var:ztr_eu_phase = 3
			has_technology_researched = digital_governance
			# Both Council-driven principles must be fully developed
			custom_tooltip = {
				text = ztr_eu_federation_principles_tt
				power_bloc ?= {
					has_principle = principle:principle_defensive_cooperation_3
					has_principle = principle:principle_foreign_investment_3
				}
			}
			custom_tooltip = {
				text = ztr_eu_federation_euro_majority_tt
				global_var:ztr_eu_tmp_euro_count >= global_var:ztr_eu_tmp_member_count
			}
			NOT = {
				has_global_variable = ztr_eu_federation_invitation_cd
			}
			any_country = {
				has_modifier = ztr_eu_member
				NOT = { this = ROOT }
			}
		}
		custom_tooltip = ztr_eu_start_federation_invitations_tt
		hidden_effect = {
			ztr_eu_start_federation_invitations = yes
		}
		remove_global_variable = ztr_eu_button_opened
		remove_global_variable = ztr_eu_tmp_member_count
		remove_global_variable = ztr_eu_tmp_euro_count
		ai_chance = {
			base = 0
		}
	}

	# Nevermind
	option = {
		name = ztr_eu_event.21.z
		default_option = yes
		remove_global_variable = ztr_eu_button_opened
		remove_global_variable = ztr_eu_tmp_member_count
		remove_global_variable = ztr_eu_tmp_euro_count
		ai_chance = {
			base = 100
		}
	}
}

# The council vote passed. Fired at every member (humans and AI): the option
# applies the political backlash of the outcome (see the effects in
# ztr_eu_scripts.txt). A passed enactment is a europeist win; a passed repeal
# is a eurosceptic one.
ztr_eu_event.25 = {
	dlc = dlc_tech_res
	type = country_event
	placement = root
	duration = 1
	title = ztr_eu_event.25.t
	desc = ztr_eu_event.25.d
	event_image = {
		video = "asia_politician_parliament_motion"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/ztr_eu.dds"
	# The outcome lands a few days after the session closes: countries that left
	# the Union in the meantime are spared the domestic fallout
	trigger = {
		has_modifier = ztr_eu_member
	}
	immediate = {
	}
	option = {
		name = ztr_eu_event.25.a
		default_option = yes
		if = {
			limit = {
				has_global_variable = ztr_eu_last_vote_proposal
				global_var:ztr_eu_last_vote_proposal = 0
			}
			ztr_eu_vote_backlash_integration_lost = yes
		}
		else = {
			ztr_eu_vote_backlash_integration_won = yes
		}
		ai_chance = {
			base = 100
		}
	}
}

# The confederal treaty conference: the chair has proclaimed the European
# Confederation and every other member state must now answer for itself.
# Accepting means adopting the Euro (if not already done) and entering the
# Confederation's power bloc; refusing means leaving the European project.
ztr_eu_event.60 = {
	dlc = dlc_tech_res
	type = country_event
	placement = root
	duration = 1
	title = ztr_eu_event.60.t
	desc = ztr_eu_event.60.d
	flavor = ztr_eu_event.60.f
	event_image = {
		video = "unspecific_signed_contract"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/ztr_eu.dds"
	trigger = {
		has_modifier = ztr_eu_member
		NOT = {
			has_variable = ztr_eu_founder
		}
	}
	immediate = {
		random_country = {
			limit = {
				has_variable = ztr_eu_founder
			}
			save_scope_as = eu_founder
		}
	}

	# Take the leap: enter the Confederation, adopting the Euro if we must
	option = {
		name = ztr_eu_event.60.a
		default_option = yes
		trigger = {
			is_not_isolationist_switzerland = yes
		}
		# Power bloc membership is incompatible with keeping subjects
		custom_tooltip = {
			text = ztr_eu_join_country_loose_all_subjects_tt
			every_subject_or_below = {
				make_independent = yes
			}
		}
		ztr_remove_any_power_bloc = yes
		# The treaty itself is the mandate: the Euro is adopted regardless of the
		# convergence criteria (this also grants the single market and bloc entry)
		if = {
			limit = {
				NOT = {
					has_modifier = ztr_euro_member
				}
			}
			custom_tooltip = ztr_eu_confed_forced_euro_tt
			hidden_effect = {
				ztr_eu_adopt_euro = yes
			}
		}
		if = {
			limit = {
				NOT = {
					has_modifier = ztr_eu_single_market
				}
			}
			add_modifier = {
				name = ztr_eu_single_market
			}
		}
		random_country = {
			limit = {
				has_variable = ztr_eu_bloc_leader
				is_power_bloc_leader = yes
			}
			root = {
				join_power_bloc = prev
			}
		}
		ai_chance = {
			base = 90
		}
	}

	# Refuse - and walk out of the European project altogether
	option = {
		name = ztr_eu_event.60.b
		ztr_eu_member_exit = yes
		ai_chance = {
			base = 10
			modifier = {
				trigger = {
					NOT = {
						ztr_eu_ideology_compatible = yes
					}
				}
				add = 25
			}
			modifier = {
				trigger = {
					ig:ig_petty_bourgeoisie = {
						is_powerful = yes
					}
				}
				add = 15
			}
		}
	}
}

	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		has_variable = ztr_eu_federation_invited

			}
		}
	}
	immediate = {
		if = {

				exists = c:EUF
			}

				save_scope_as = eu_federation_partner
			}
		}
		else = {
			random_country = {
				limit = {
					has_variable = ztr_eu_federation_proposer
				}
				save_scope_as = eu_federation_partner
			}
		}
	}

	# Accept: the player country always survives the merger; AI countries join the
	# proposer or, after the first acceptance, the already existing Federation
	option = {
		name = ztr_eu_event.70.a
		trigger = {
			is_not_isolationist_switzerland = yes
		}
		custom_tooltip = ztr_eu_federation_ratify_tt
		hidden_effect = {
			remove_variable = ztr_eu_federation_invited
			ztr_eu_accept_federation_invitation = yes
		}
		ai_chance = {
			base = 55
			modifier = {
				trigger = {
					ztr_eu_ideology_compatible = yes
					"relations(scope:eu_federation_partner)" >= relations_threshold:amicable
				}
				add = 30
			}
			modifier = {
				trigger = {
					has_modifier = ztr_euro_member
				}
				add = 10
			}
			modifier = {
				trigger = {
					country_rank = rank_value:great_power
				}
				add = -25
			}
			modifier = {
				trigger = {
					ig:ig_petty_bourgeoisie = {
						is_powerful = yes
					}
				}
				add = -25
			}
		}
	}

	# Refuse: remain independent; the post-federation accession decision stays open
	option = {
		name = ztr_eu_event.70.b
		default_option = yes
		custom_tooltip = ztr_eu_federation_refuse_tt
		hidden_effect = {
			remove_variable = ztr_eu_federation_invited
			if = {
				limit = {
					exists = scope:eu_federation_partner
				}
				change_relations = {
					country = scope:eu_federation_partner
					value = -25
				}
			}
		}
		ai_chance = {
			base = 45
		}
	}
