namespace = ztr_eu_event

# ===================================================================================
# The European Project
# 1    EU founded (chair)                    2  invitation to EEC members
# 10   council vote opened (humans)          11 parliament reapportioned
# 20   directives menu                       21 treaties menu (chair)
# 25   vote passed                           26 vote failed
#      (25/26 go to every member, humans and AI: they carry the political backlash)
# 30   Euro adopted                          31 ECB established
# 32   ECB stance changed
# 40   the Union's political orientation shifted (decennial review)
# 60   confederal treaty conference (every member decides for itself)
# 70   individual invitation to federal union
# (41-42 ideological drift and 61 eurozone-gated market were removed)
# ===================================================================================

# EU foundation - the chair signs the Treaty on European Union
ztr_eu_event.1 = {
	dlc = dlc_tech_res
	type = country_event
	placement = root

	duration = 1

	title = ztr_eu_event.1.t
	desc = ztr_eu_event.1.d
	flavor = ztr_eu_event.1.f

	event_image = {
		video = "unspecific_signed_contract"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/ztr_eu.dds"

	immediate = {
		save_scope_as = eu_founder
	}

	option = {
		name = ztr_eu_event.1.a
		default_option = yes
		change_infamy = -5
		ztr_eu_member_accession = yes
		hidden_effect = {
			every_country = {
				limit = {
					NOT = { this = root }
					has_modifier = ztr_eec_member
				}
				trigger_event = {
					id = ztr_eu_event.2
					popup = yes
				}
			}
		}
		ai_chance = {
			base = 100
		}
	}
}

# EU foundation - the other EEC members decide whether to follow
ztr_eu_event.2 = {
	dlc = dlc_tech_res
	type = country_event
	placement = root

	duration = 1

	title = ztr_eu_event.2.t
	desc = ztr_eu_event.2.d
	flavor = ztr_eu_event.1.f

	event_image = {
		video = "unspecific_signed_contract"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/ztr_eu.dds"

	immediate = {
		random_country = {
			limit = {
				has_variable = ztr_eu_founder
			}
			save_scope_as = eu_founder
		}
	}

	# Join the Union
	option = {
		name = ztr_eu_event.2.a
		default_option = yes
		trigger = {
			is_not_isolationist_switzerland = yes
		}
		# If a Country joins the EU, all of its European subjects will become independent.
		# Overseas dependencies are kept: the EEC founders joined while holding colonial empires.
		custom_tooltip = {
			text = ztr_eu_join_country_loose_all_subjects_tt
			every_subject_or_below = {
				limit = { ztr_is_european_country = yes }
				make_independent = yes
			}
		}
		change_infamy = -5
		ztr_eu_member_accession = yes
		ai_chance = {
			base = 80
			modifier = {
				trigger = {
					NOT = {
						ztr_eu_ideology_compatible = yes
					}
				}
				add = -60
			}
			modifier = {
				trigger = {
					"relations(scope:eu_founder)" >= relations_threshold:amicable
				}
				add = 20
			}
		}
	}

	# Stay out: the EEC dissolves beneath our feet
	option = {
		name = ztr_eu_event.2.b
		hidden_effect = {
			if = {
				limit = {
					has_modifier = ztr_eec_member
				}
				remove_modifier = ztr_eec_member
			}
		}
		every_country = {
			limit = {
				has_modifier = ztr_eu_member
			}
			change_relations = {
				country = root
				value = -20
			}
		}
		ai_chance = {
			base = 20
			modifier = {
				trigger = {
					NOT = {
						ztr_eu_ideology_compatible = yes
					}
				}
				add = 60
			}
		}
	}
}
