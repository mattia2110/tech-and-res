# Weekly mandate progress of the European Confederation power bloc: always zero.
# The Union's principles advance only through Council votes, never through mandates.
ztr_eu_zero_mandate_progress_sv = {
	value = 0
}

# European Parliament seat apportionment.
# 360 seats split among member states proportionally to population,
# with a floor of 6 seats per country (degressive proportionality).
# Requires global_var:ztr_eu_total_member_pop to be set before evaluation.
ztr_eu_country_seats_sv = {
	value = total_population
	multiply = 360
	divide = global_var:ztr_eu_total_member_pop
	round = yes
	min = 6
}

# Europeist share of a member's parliament delegation during a Council vote.
# Requires var:ztr_eu_seats (seat snapshot) and var:ztr_eu_j_pop (popular
# europeanism, 0..1, set by ztr_eu_compute_europeanism) on the country.
ztr_eu_europeist_seats_sv = {
	value = var:ztr_eu_seats
	multiply = var:ztr_eu_j_pop
	round = yes
	min = 0
}
