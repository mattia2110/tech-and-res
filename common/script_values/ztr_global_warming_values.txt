# Global Warming - climate disaster tuning constants
#
# The whole climate-disaster mechanic is driven by the global variable
# ztr_global_warming_temperature_anomaly, expressed in degrees Celsius above the
# pre-industrial baseline on a 0-6 scale (see ztr_global_warming_scripts.txt).

### Temperature model ###
#
# WARNING: the four constants in this section and the four stage thresholds
# below are REFERENCE VALUES ONLY. They are not read at runtime.
#
# change_global_variable only accepts a variable or a number for its operands,
# and a global_var comparison only accepts a variable or a number on the right
# hand side, so the actual call sites had to be written as literals:
#
#   divisor and decay  -> scripted_effects/ztr_global_warming_scripts.txt
#   ceiling            -> scripted_effects/ztr_global_warming_scripts.txt
#   stage thresholds   -> scripted_triggers/ztr_main_triggers.txt
#
# Editing a number here changes nothing on its own. Change it in the file above
# as well, or the documentation and the behaviour will drift apart.
#
# Only ztr_gw_hc_chance_mult and ztr_gw_hc_intensity_bonus, at the bottom of
# this file, are genuinely evaluated at runtime.

# Yearly temperature growth is global pollution divided by this value.
# The pollution variable is already at 1/100 of the raw pollution_generation
# scale, so the effective rate is raw_pollution / (100 * this).
#
# Because the model also decays 1% of the accumulated anomaly per year, the
# equilibrium temperature is:
#
#     T_eq = 100 * ztr_global_warming_total_pollution_var / ztr_gw_temperature_divisor
#
# TUNING: the divisor below assumes a late-game global pollution variable of
# roughly 4500 (raw pollution ~450000), which yields an equilibrium of ~3.0 C.
# The journal entry displays both the current pollution and the projected
# equilibrium, so this can be re-derived from an actual campaign with:
#
#     new_divisor = 100 * observed_pollution_var / desired_equilibrium
ztr_gw_temperature_divisor = 150000

# Fraction of the accumulated anomaly reabsorbed each year.
ztr_gw_temperature_decay_rate = 0.01

# Hard ceiling of the anomaly. Matches the 0-6 scale of the journal entry
# progress bar (max_value 600, fed with anomaly * 100).
ztr_gw_temperature_max = 6.0

### Climate stage thresholds (degrees Celsius) ###

ztr_gw_stage_1_threshold = 0.5   # Pollution-driven warming is recognized
ztr_gw_stage_2_threshold = 1.0   # Glacial retreat and extreme heat become widespread
ztr_gw_stage_3_threshold = 1.5   # Sea-level rise becomes a global crisis
ztr_gw_stage_4_threshold = 2.0   # Point of no return: a fundamentally altered Earth

### Harvest condition scaling ###

# Multiplier applied to the chance of climate-sensitive conditions.
# 1.0 at 0 C (no change), 7.0 at 6 C.
ztr_gw_hc_chance_mult = {
	value = 1
	if = {
		limit = {
			has_game_rule = allow_ztr_climate_disasters
			ztr_global_warming_enabled = yes
			has_global_variable = ztr_global_warming_temperature_anomaly
		}
		add = global_var:ztr_global_warming_temperature_anomaly
	}
}

# Intensity added to climate conditions in discrete climate-stage steps.
# Stage 0 and stage 1 add 0; stages 2, 3 and 4 add 1, 2 and 3 respectively.
ztr_gw_hc_intensity_bonus = {
	value = 0
	if = {
		limit = {
			has_game_rule = allow_ztr_climate_disasters
			ztr_global_warming_stage_4 = yes
		}
		add = 3
	}
	else_if = {
		limit = {
			has_game_rule = allow_ztr_climate_disasters
			ztr_global_warming_stage_3 = yes
		}
		add = 2
	}
	else_if = {
		limit = {
			has_game_rule = allow_ztr_climate_disasters
			ztr_global_warming_stage_2 = yes
		}
		add = 1
	}
}
