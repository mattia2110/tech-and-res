# 0 (rainforest, -50%) to 7 (desert, +33%); 3 is neutral geography. Needs ztr_mm_renewable_value_var.
ztr_mm_renewable_level_calc = {
	value = 3
	if = {
		limit = { var:ztr_mm_renewable_value_var < -0.3 }
		value = 0
	}
	else_if = {
		limit = { var:ztr_mm_renewable_value_var < -0.15 }
		value = 1
	}
	else_if = {
		limit = { var:ztr_mm_renewable_value_var < -0.05 }
		value = 2
	}
	else_if = {
		limit = { var:ztr_mm_renewable_value_var < 0.05 }
		value = 3
	}
	else_if = {
		limit = { var:ztr_mm_renewable_value_var < 0.125 }
		value = 4
	}
	else_if = {
		limit = { var:ztr_mm_renewable_value_var < 0.2 }
		value = 5
	}
	else_if = {
		limit = { var:ztr_mm_renewable_value_var < 0.3 }
		value = 6
	}
	else = {
		value = 7
	}
}

# Cached yearly by ztr_mm_update_map_modes.
ztr_mm_renewable_value = {
	value = 0
	if = {
		limit = { has_variable = ztr_mm_renewable_value_var }
		value = var:ztr_mm_renewable_value_var
	}
}

ztr_mm_renewable_level = {
	value = -1
	if = {
		limit = { has_variable = ztr_mm_renewable_level_var }
		value = var:ztr_mm_renewable_level_var
	}
}

#### Innovation ####################

# Leader stored yearly by ztr_mm_update_map_modes; the GUI compares live weekly innovation against it.
ztr_mm_innovation_ready = {
	value = 0
	if = {
		limit = { has_global_variable = ztr_mm_innovation_leader_var }
		value = 1
	}
}

#### Liberty and Equality Index ####################

ztr_mm_liberty_value = {
	value = 0
	if = {
		limit = { owner ?= { has_variable = liberty_index_var } }
		value = owner.var:liberty_index_var
	}
}

ztr_mm_equality_value = {
	value = 0
	if = {
		limit = { owner ?= { has_variable = equality_index_var } }
		value = owner.var:equality_index_var
	}
}

# Index 0-100 split in 8 steps.
ztr_mm_liberty_level = {
	value = -1
	if = {
		limit = { owner ?= { has_variable = liberty_index_var } }
		value = ztr_mm_liberty_value
		divide = 12.5
		floor = yes
		min = 0
		max = 7
	}
}

ztr_mm_equality_level = {
	value = -1
	if = {
		limit = { owner ?= { has_variable = equality_index_var } }
		value = ztr_mm_equality_value
		divide = 12.5
		floor = yes
		min = 0
		max = 7
	}
}

#### Demographic Stage ####################

ztr_mm_demography_level = {
	value = -1
	if = {
		limit = { exists = owner }
		value = owner.ztr_demography_applied_stage
	}
}

#### Nuclear Powers ####################

# 0 none, 1 civil (nuclear_reactor + civil or military uranium law), 2 military (nuclear_capability + military uranium law).
ztr_mm_nuclear_level = {
	value = 0
	if = {
		limit = {
			owner ?= {
				has_modifier = nuclear_capability
				has_law = law_type:law_uranium_full_usage
			}
		}
		value = 2
	}
	else_if = {
		limit = {
			owner ?= {
				has_technology_researched = nuclear_reactor
				OR = {
					has_law = law_type:law_uranium_civil_usage
					has_law = law_type:law_uranium_full_usage
				}
			}
		}
		value = 1
	}
}
