# Research agreements and their points in je_research_center
je_research_center_agreements_sgui = {
	scope = country

	effect = {
		if = {
			limit = {
				ztr_research_agreements_num > 0
			}
			every_scope_treaty = {
				limit = {
					any_scope_article = { has_type = ztr_research_agreement }
				}
				if = {
					limit = { first_country = root }
					second_country = { save_temporary_scope_as = ztr_ra_partner }
				}
				else = {
					first_country = { save_temporary_scope_as = ztr_ra_partner }
				}
				save_temporary_scope_value_as = {
					name = ztr_ra_partner_points
					value = root.ztr_research_agreement_weight
				}
				custom_tooltip_no_bullet = "JE_RESEARCH_CENTER_AGREEMENT_ENTRY"
			}
			if = {
				limit = {
					ztr_research_agreements_over_cap > 0
				}
				custom_tooltip_no_bullet = "JE_RESEARCH_CENTER_AGREEMENTS_CAP"
			}
		}
		else = {
			custom_tooltip_no_bullet = "JE_RESEARCH_CENTER_NO_AGREEMENTS"
		}
	}
}
