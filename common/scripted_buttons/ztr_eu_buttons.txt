# European Council and ECB buttons

# Open the directive proposal menu (any member state may table a proposal)
ztr_eu_open_directives_menu = {
	name = "ztr_eu_open_directives_menu"
	desc = "ztr_eu_open_directives_menu_desc"
	visible = {
		has_journal_entry = je_eu_info_panel
	}
	possible = {
		custom_tooltip = {
			text = "ztr_eu_vote_session_clear_tt"
			NOT = {
				has_global_variable = ztr_eu_vote_active
			}
		}
		custom_tooltip = {
			text = "ztr_eu_vote_cooldown_tt"
			NOT = {
				has_global_variable = ztr_eu_vote_cd
			}
		}
		has_modifier = ztr_eu_member
		hidden_trigger = {
			is_player = yes
			NOT = {
				has_global_variable = ztr_eu_button_opened
			}
		}
	}
	effect = {
		set_global_variable = ztr_eu_button_opened
		trigger_event = {
			id = ztr_eu_event.20
			popup = yes
		}
	}
	ai_chance = {
		value = 0
	}
}

# Voting buttons on the council vote journal entry
ztr_eu_vote_for_statement = {
	name = "ztr_eu_vote_for_statement"
	desc = "ztr_eu_vote_for_statement_desc"
	visible = {
		is_involved_in_journal_entry = je_eu_votations
		# Only sitting members with a ballot may vote: countries that left the Union
		# mid-session stay involved in the journal entry but lose their vote
		has_modifier = ztr_eu_member
		has_variable = ztr_eu_vote_eligible
		global_var:ztr_eu_vote_months_remaining > 0
	}
	possible = {
		NOT = {
			has_variable = ztr_eu_vote_choice_support
		}
	}
	effect = {
		if = {
			limit = {
				has_variable = ztr_eu_vote_choice_against
			}
			remove_variable = ztr_eu_vote_choice_against
		}
		set_variable = ztr_eu_vote_choice_support
		je:je_eu_votations = {
			ztr_eu_vote_update_results = yes
		}
	}
	ai_chance = {
		value = 0
	}
}
