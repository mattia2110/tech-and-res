war_nuclear_strike = {
	groups= { general }
	requires_approval = no
	should_notify_third_parties = yes

	unlocking_technologies = {
		atomic_bomb
	}

	potential = {
		#THIS has nuclear weapons
		has_modifier = nuclear_capability
		# Cheap checks first: both branches below require TARGET to be at war
		scope:target_country = {
			is_at_war = yes
			is_revolutionary = no
			is_secessionist = no
		}
		NOT = { is_in_war_together = scope:target_country }
		#Civil war countries do not nuke each other
		is_revolutionary = no is_secessionist = no
		#THIS is at war OR a country that THIS has stationed nuclear weapons in is at war
		OR = {
			is_at_war = yes
			any_country = {
				has_diplomatic_pact = { who = scope:actor type = station_nuclear_weapons is_initiator = no }
				is_at_war = yes
				any_scope_war = {
					is_warleader = prev
				}
			}
		}
		#Can only threaten an enemy war leader
		OR = {
			AND = {
				has_war_with = scope:target_country
				scope:target_country = { is_diplomatic_play_enemy_of = scope:actor }
				any_scope_war = {
					is_warleader = scope:actor
					is_warleader = scope:target_country
					is_war_participant = scope:actor
					is_war_participant = scope:target_country
				}
			}
			any_country = {
				has_diplomatic_pact = { who = scope:actor type = station_nuclear_weapons is_initiator = no }
				has_war_with = scope:target_country
				any_scope_war = {
					is_warleader = prev
					is_warleader = scope:target_country
				}
				is_diplomatic_play_enemy_of = scope:target_country
			}
		}
	}

	possible = {
		ztr_can_launch_nuclear_strike = { TARGET = scope:target_country }
	}

	accept_effect = {
		if = {
			limit = { ztr_can_launch_nuclear_strike = { TARGET = scope:target_country } }
			#Lobby
			add_lobby_appeasement_from_diplomacy_unidirectional = { 
				FIRST = root
				SECOND = scope:target_country 
				PRO_AMOUNT = -10
				ANTI_AMOUNT = 5
				FACTOR = appeasement_embargo_declared
			}
			#THIS gains infamy and international hatred for unprovoked nuclear attack
			create_incident = {
				value = 50
				country = scope:target_country
			}
			ztr_launch_nuclear_exchange = {
				ATTACKER = scope:actor
				DEFENDER = scope:target_country
				ESCALATE = no
			}
		}
	}
	
	ai = {
		evaluation_chance = {
			value = 0.25		
		}
		propose_score = {
			value = 0
		}
		will_propose = {
			# This would be an hell to balance properly, so AI can use this only if is the first time that nuclear weapons are used in the world
			NOT = { has_global_variable = ztr_atomic_bomb_first_use_var }
		}
	}
}
