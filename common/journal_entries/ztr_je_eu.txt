# The European Project journal entries

# Union overview panel: visible to every member from the EU phase onwards
je_eu_info_panel = {
	icon = "gfx/interface/icons/event_icons/ztr_eu.dds"
	group = je_group_foreign_affairs
	is_shown_when_inactive = {
		has_modifier = ztr_eu_member
		has_global_variable = ztr_eu_created
		global_var:ztr_eu_phase < 4
	}
	possible = {
		has_modifier = ztr_eu_member
		has_global_variable = ztr_eu_created
		global_var:ztr_eu_phase < 4
	}
	immediate = {
		ztr_eu_ensure_chair_exists = yes
		if = {
			limit = {
				NOT = {
					has_variable = ztr_eu_seats
				}
			}
			ztr_eu_distribute_seats = yes
		}
		ztr_eu_member_ranking_variables = yes
		# First europeanism reading, so the panel never shows an empty value
		if = {
			limit = {
				NOT = {
					has_variable = ztr_eu_j_pop
				}
			}
			ztr_eu_compute_europeanism = yes
		}
	}
	on_yearly_pulse = {
		effect = {
			ztr_eu_ensure_chair_exists = yes
			# The chair rotates annually to the most prestigious member
			ztr_eu_rotate_chair = yes
			ztr_eu_member_ranking_variables = yes
			# Yearly europeanism reading: keeps the panel fresh and feeds the AI lean
			ztr_eu_compute_europeanism = yes
			# Chair-side bookkeeping, executed once for the whole Union
			if = {
				limit = {
					has_variable = ztr_eu_founder
				}
				# Five-year legislature: reapportion the 360 seats
				if = {
					limit = {
						has_global_variable = ztr_eu_seats_timer
					}
					change_global_variable = {
						name = ztr_eu_seats_timer
						add = -1
					}
					if = {
						limit = {
							global_var:ztr_eu_seats_timer <= 0
						}
						ztr_eu_distribute_seats = yes
						set_global_variable = {
							name = ztr_eu_seats_timer
							value = 5
						}
						every_country = {
							limit = {
								has_modifier = ztr_eu_member
								is_ai = no
							}
							trigger_event = {
								id = ztr_eu_event.11
								days = 3
							}
						}
					}
				}
				else = {
					set_global_variable = {
						name = ztr_eu_seats_timer
						value = 5
					}
				}
				# Decennial review of the Union's political orientation (EU phase
				# only: the orientation freezes once the Confederation exists)
				if = {
					limit = {
						global_var:ztr_eu_phase = 2
					}
					if = {
						limit = {
							has_global_variable = ztr_eu_ideology_timer
						}
						change_global_variable = {
							name = ztr_eu_ideology_timer
							add = -1
						}
						if = {
							limit = {
								global_var:ztr_eu_ideology_timer <= 0
							}
							# Hysteresis: a challenger family must outnumber the
							# incumbent by 2 to flip the routine review
							set_global_variable = {
								name = ztr_eu_ideology_review_margin
								value = 2
							}
							ztr_eu_review_union_ideology = yes
							set_global_variable = {
								name = ztr_eu_ideology_timer
								value = 10
							}
						}
					}
					else = {
						set_global_variable = {
							name = ztr_eu_ideology_timer
							value = 10
						}
					}
				}
				# AI chair tables proposals when the floor is free
				if = {
					limit = {
						is_ai = yes
						NOR = {
							has_global_variable = ztr_eu_vote_active
							has_global_variable = ztr_eu_vote_cd
							has_global_variable = ztr_eu_button_opened
						}
					}
					random = {
						chance = 40
						ztr_eu_ai_propose_directive = yes
					}
				}
			}
			# AI members adopt the Euro once they qualify
			if = {
				limit = {
					is_ai = yes
					has_modifier = ztr_eu_member
					NOT = {
						has_modifier = ztr_euro_member
					}
					ztr_eu_euro_convergence_criteria = yes
				}
				random = {
					chance = 25
					ztr_eu_adopt_euro = yes
				}
			}
		}
	}
	scripted_button = ztr_eu_open_directives_menu
	scripted_button = ztr_eu_open_treaties_menu
	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_eu_info_panel_status_chair
				trigger = {
					has_variable = ztr_eu_founder
				}
			}
			triggered_desc = {
				desc = je_eu_info_panel_status_single_market
				trigger = {
					has_modifier = ztr_eu_single_market
				}
			}
			triggered_desc = {
				desc = je_eu_info_panel_status_member
				trigger = {
					always = yes
				}
			}
		}
	}
	invalid = {
		OR = {
			NOT = {
				has_modifier = ztr_eu_member
			}
			global_var:ztr_eu_phase >= 4
		}
	}
	weight = 100
	can_deactivate = yes
	transferable = no
	progressbar = no
}

# Council vote on a directive or treaty, resolved by double majority
je_eu_votations = {
	icon = "gfx/interface/icons/event_icons/ztr_eu.dds"
	group = je_group_global_eu_content
	is_shown_when_inactive = {
		has_global_variable = ztr_eu_vote_active
	}
	should_be_involved = {
		has_variable = ztr_eu_vote_eligible
		has_modifier = ztr_eu_member
	}
	should_show_when_not_involved = {
		has_modifier = ztr_eu_member
	}
	is_shown_in_lobby = {
		has_global_variable = ztr_eu_vote_active
		has_modifier = ztr_eu_member
	}
	possible = {
		has_global_variable = ztr_eu_vote_active
	}
	immediate = {
		scope:journal_entry = {
			ztr_eu_vote_update_ai_positions = yes
			ztr_eu_vote_update_results = yes
		}
	}
	on_monthly_pulse = {
		effect = {
			scope:journal_entry = {
				ztr_eu_vote_update_ai_positions = yes
				ztr_eu_vote_update_results = yes
			}
			# Countdown: decrement months remaining
			if = {
				limit = {
					has_global_variable = ztr_eu_vote_months_remaining
					global_var:ztr_eu_vote_months_remaining > 0
				}
				change_global_variable = {
					name = ztr_eu_vote_months_remaining
					add = -1
				}
			}
			# Session over: resolve by double majority
			if = {
				limit = {
					has_global_variable = ztr_eu_vote_months_remaining
					global_var:ztr_eu_vote_months_remaining <= 0
				}
				scope:journal_entry = {
					ztr_eu_vote_update_ai_positions = yes
					ztr_eu_vote_update_results = yes
				}
				# Preserve the subject for the outcome events, which fire after cleanup
				if = {
					limit = {
						has_global_variable = ztr_eu_vote_subject
					}
					set_global_variable = {
						name = ztr_eu_last_vote_subject
						value = global_var:ztr_eu_vote_subject
					}
				}
				if = {
					limit = {
						has_global_variable = ztr_eu_vote_proposal
					}
					set_global_variable = {
						name = ztr_eu_last_vote_proposal
						value = global_var:ztr_eu_vote_proposal
					}
				}
				if = {
					limit = {
						ztr_eu_vote_double_majority_passes = yes
					}
					ztr_eu_apply_council_vote = yes
					# Every member receives the outcome: the event applies the
					# political backlash, so the AI must get it too
					every_country = {
						limit = {
							has_modifier = ztr_eu_member
						}
						trigger_event = {
							id = ztr_eu_event.25
							days = 3
						}
					}
				}
				else = {
					every_country = {
						limit = {
							has_modifier = ztr_eu_member
						}
						trigger_event = {
							id = ztr_eu_event.26
							days = 3
						}
					}
					if = {
						limit = {
							has_global_variable = ztr_eu_vote_subject
						}
						remove_global_variable = ztr_eu_vote_subject
					}
					if = {
						limit = {
							has_global_variable = ztr_eu_vote_proposal
						}
						remove_global_variable = ztr_eu_vote_proposal
					}
				}
				ztr_eu_vote_cleanup_all_members = yes
				# Two years before the Council reconvenes
				set_global_variable = {
					name = ztr_eu_vote_cd
					years = 2
				}
			}
			# Edge case: nobody left to vote
			else_if = {
				limit = {
					NOT = {
						any_country = {
							OR = {
								has_variable = ztr_eu_vote_eligible
								is_involved_in_journal_entry = je_eu_votations
							}
						}
					}
				}
				ztr_eu_vote_cleanup_all_members = yes
			}
		}
	}
	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_eu_votations_status_passing
				trigger = {
					ztr_eu_vote_double_majority_passes = yes
				}
			}
			triggered_desc = {
				desc = je_eu_votations_status_failing
				trigger = {
					always = yes
				}
			}
		}
	}
	scripted_progress_bar = ztr_eu_vote_support_progress_bar
	scripted_progress_bar = ztr_eu_vote_against_progress_bar
	scripted_button = ztr_eu_vote_for_statement
	scripted_button = ztr_eu_vote_against_statement
	weight = 110
	should_be_pinned_by_default_uninvolved_or_context = yes
	can_deactivate = yes
	transferable = no
	progressbar = no
}
