je_global_warming = {
	icon = "gfx/interface/icons/event_icons/global_warming.dds"
	group = je_group_global_historical_content
	scripted_progress_bar = ztr_global_warming_temperature_progress_bar

	should_be_involved = {
		ztr_global_warming_enabled = yes
	}

	is_shown_when_inactive = {
		ztr_global_warming_enabled = yes
	}

	possible = {
		ztr_global_warming_enabled = yes
	}

	# The two blocks below are guarded as well, so the temperature model cannot be
	# initialised even if the entry is somehow instantiated for a single tick.
	immediate = {
		if = {
			limit = {
				ztr_global_warming_enabled = yes
			}
			ztr_global_warming_yearly_update = yes

			scope:journal_entry = {
				set_bar_progress = {
					value = {
						value = global_var:ztr_global_warming_temperature_anomaly
						multiply = 100
					}
					name = ztr_global_warming_temperature_progress_bar
				}
			}
		}
	}

	on_yearly_pulse = {
		effect = {
			if = {
				limit = {
					ztr_global_warming_enabled = yes
				}
				ztr_global_warming_yearly_update = yes

				scope:journal_entry = {
					set_bar_progress = {
						value = {
							value = global_var:ztr_global_warming_temperature_anomaly
							multiply = 100
						}
						name = ztr_global_warming_temperature_progress_bar
					}
				}
			}
		}
	}


	status_desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					ztr_global_warming_stage_4 = yes
				}
				desc = je_global_warming_status_critical
			}
			triggered_desc = {
				trigger = {
					ztr_global_warming_stage_3 = yes
				}
				desc = je_global_warming_status_severe
			}
			triggered_desc = {
				trigger = {
					ztr_global_warming_stage_2 = yes
				}
				desc = je_global_warming_status_established
			}
			triggered_desc = {
				trigger = {
					ztr_global_warming_stage_1 = yes
				}
				desc = je_global_warming_status_warming
			}
			desc = je_global_warming_status
		}
	}

	weight = 1000
	transferable = no
	should_be_pinned_by_default_involved = yes
	should_be_pinned_by_default_uninvolved_or_context = yes
}
