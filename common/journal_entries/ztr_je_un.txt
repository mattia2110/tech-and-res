je_un_info_panel = {
	icon = "gfx/interface/icons/event_icons/united_nations.dds"
	group = je_group_global_un_content
	context = country

	activation_description = {
		always = yes
	}
	is_shown_when_inactive = {
		ztr_is_un_member = yes
	}
	possible = {
		ztr_is_un_member = yes
	}
	# UN institutions: 1 while established, 0 otherwise
	scaling_modifiers_while_active = {
		modifier = ztr_wto_member_modifier
		value = {
			value = 0
			if = {
				limit = { ztr_is_un_institution_member = { INSTITUTION = wto } }
				value = 1
			}
		}
	}
	scaling_modifiers_while_active = {
		modifier = ztr_icc_member_modifier
		value = {
			value = 0
			if = {
				limit = { ztr_is_un_institution_member = { INSTITUTION = icc } }
				value = 1
			}
		}
	}
	scaling_modifiers_while_active = {
		modifier = ztr_unesco_member_modifier
		value = {
			value = 0
			if = {
				limit = { ztr_is_un_institution_member = { INSTITUTION = unesco } }
				value = 1
			}
		}
	}
	# UN dues: weekly expense fixed at the start of the year (ztr_un_assess_dues), none while suspended
	scaling_modifiers_while_active = {
		modifier = ztr_un_dues_modifier
		value = {
			value = ztr_un_dues_weekly_paid
		}
	}
	# Yearly AI proposals: ztr_un_ai_yearly_proposal in un_institutions_on_yearly_pulse
	scripted_button = ztr_un_open_governance_menu
	scripted_button = ztr_un_open_human_rights_menu
	scripted_button = ztr_un_open_institutions_menu
	scripted_button = ztr_un_open_elected_human_rights_menu
	scripted_button = ztr_un_seat_candidacy
	scripted_button = ztr_un_withhold_dues
	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_un_info_panel_status_suspended
				trigger = {
					ztr_un_is_suspended = yes
				}
			}
			triggered_desc = {
				desc = je_un_info_panel_status_permanent
				trigger = {
					has_modifier = ztr_onu_permanent_member
				}
			}
			triggered_desc = {
				desc = je_un_info_panel_status_elected
				trigger = {
					ztr_un_is_elected_member = yes
				}
			}
			triggered_desc = {
				desc = je_un_info_panel_status_member
				trigger = {
					has_modifier = ztr_onu_member
				}
			}
		}
	}
	# Map: permanent members dark blue, elected members mid blue, member states light blue
	map_color = {
		if = {
			limit = {
				owner ?= { has_modifier = ztr_onu_permanent_member }
			}
			value = "ztr_un_permanent_member_blue"
		}
		else_if = {
			limit = {
				owner ?= { ztr_un_is_elected_member = yes }
			}
			value = "ztr_un_elected_member_blue"
		}
		else_if = {
			limit = {
				owner ?= { has_modifier = ztr_onu_member }
			}
			value = "ztr_un_member_blue"
		}
	}
	map_tooltip = {
		if = {
			limit = {
				owner ?= { has_modifier = ztr_onu_permanent_member }
			}
			custom_tooltip_no_bullet = "ztr_un_map_permanent_member_tt"
		}
		else_if = {
			limit = {
				owner ?= { ztr_un_is_elected_member = yes }
			}
			custom_tooltip_no_bullet = "ztr_un_map_elected_member_tt"
		}
		else_if = {
			limit = {
				owner ?= { has_modifier = ztr_onu_member }
			}
			custom_tooltip_no_bullet = "ztr_un_map_member_tt"
		}
		else = {
			custom_tooltip_no_bullet = "ztr_un_map_not_member_tt"
		}
	}
	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			NOT = {
				ztr_is_un_member = yes
			}
		}
	}
	weight = 100
	can_deactivate = yes
	transferable = no
	progressbar = no
}

je_un_votations = {
	icon = "gfx/interface/icons/event_icons/united_nations.dds"
	group = je_group_global_un_content

	activation_description = {
		always = yes
	}
	is_shown_when_inactive = {
		has_global_variable = ztr_un_vote_active
	}
	should_be_involved = {
		has_variable = ztr_un_vote_eligible
	}
	should_show_when_not_involved = {
		ztr_is_un_member = yes
	}
	is_shown_in_lobby = {
		has_global_variable = ztr_un_vote_active
		ztr_is_un_member = yes
	}
	possible = {
		has_global_variable = ztr_un_vote_active
	}
	immediate = {
		scope:journal_entry = {
			ztr_un_vote_update_neutral_ai_positions = yes
			ztr_un_vote_update_results = yes
		}
	}
	on_monthly_pulse = {
		effect = {
			scope:journal_entry = {
				ztr_un_vote_update_neutral_ai_positions = yes
				ztr_un_vote_update_results = yes
			}
			# Countdown: decrement months remaining
			if = {
				limit = {
					has_global_variable = ztr_un_vote_months_remaining
					global_var:ztr_un_vote_months_remaining > 0
				}
				change_global_variable = {
					name = ztr_un_vote_months_remaining
					add = -1
				}
			}
			# Vote session ended: resolve and deactivate
			if = {
				limit = {
					has_global_variable = ztr_un_vote_months_remaining
					global_var:ztr_un_vote_months_remaining <= 0
				}
				scope:journal_entry = {
					ztr_un_vote_update_neutral_ai_positions = yes
					ztr_un_vote_update_results = yes
				}
				# Result (adopted, rejected or vetoed) kept for the result event
				ztr_un_vote_save_last_result = yes
				if = {
					limit = {
						global_var:ztr_un_last_vote_result = 1
					}
					# Vote passed - apply the governance change
					ztr_un_apply_governance_vote = yes
				}
				else = {
					# Vote failed (majority lost or vetoed) - clean up subject/proposal
					if = {
						limit = {
							has_global_variable = ztr_un_vote_subject
						}
						remove_global_variable = ztr_un_vote_subject
					}
					if = {
						limit = {
							has_global_variable = ztr_un_vote_proposal
						}
						remove_global_variable = ztr_un_vote_proposal
					}
				}
				# Cleanup removes ztr_un_vote_active, making possible = no, then JE deactivates
				ztr_un_vote_cleanup_all_members = yes
				every_country = {
					limit = {
						is_ai = no
						ztr_is_un_member = yes
					}
					trigger_event = {
						id = ztr_un_event.11
						days = 0
						popup = yes
					}
				}
				# Cooldown starts: set cooldown variable for 4 years (48 months)
				set_global_variable = {
					name = ztr_un_vote_cd
					years = 4
				}
			}
			# Edge case: no eligible countries left, then deactivate
			else_if = {
				limit = {
					NOT = {
						any_country = {
							OR = {
								has_variable = ztr_un_vote_eligible
								is_involved_in_journal_entry = je_un_votations
							}
						}
					}
				}
				ztr_un_vote_cleanup_all_members = yes
			}
		}
	}
	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_un_votations_status_support
				trigger = {
					global_var:ztr_un_vote_support_count > global_var:ztr_un_vote_against_count
				}
			}
			triggered_desc = {
				desc = je_un_votations_status_against
				trigger = {
					global_var:ztr_un_vote_against_count > global_var:ztr_un_vote_support_count
				}
			}
			triggered_desc = {
				desc = je_un_votations_status_neutral
				trigger = {
					always = yes
				}
			}
		}
	}
	# Map: vote of each eligible member; gold stripes on permanent members that cast a veto
	map_color = {
		if = {
			limit = {
				owner ?= { has_variable = ztr_un_vote_choice_support }
			}
			value = "ztr_un_vote_support_green"
		}
		else_if = {
			limit = {
				owner ?= { has_variable = ztr_un_vote_choice_against }
			}
			value = "ztr_un_vote_against_red"
		}
		else_if = {
			limit = {
				owner ?= { has_variable = ztr_un_vote_eligible }
			}
			value = "ztr_un_vote_neutral_white"
		}
	}
	map_color_second = {
		if = {
			limit = {
				global_var:ztr_un_law_veto_right ?= 1
				owner ?= {
					has_variable = ztr_un_vote_veto
				}
			}
			value = "ztr_un_vote_veto_gold"
		}
	}
	map_tooltip = {
		if = {
			limit = {
				owner ?= { has_variable = ztr_un_vote_choice_support }
			}
			custom_tooltip_no_bullet = "ztr_un_vote_map_support_tt"
		}
		else_if = {
			limit = {
				owner ?= { has_variable = ztr_un_vote_choice_against }
			}
			custom_tooltip_no_bullet = "ztr_un_vote_map_against_tt"
		}
		else_if = {
			limit = {
				owner ?= { has_variable = ztr_un_vote_eligible }
			}
			custom_tooltip_no_bullet = "ztr_un_vote_map_neutral_tt"
		}
		else = {
			custom_tooltip_no_bullet = "ztr_un_vote_map_not_eligible_tt"
		}
		if = {
			limit = {
				global_var:ztr_un_law_veto_right ?= 1
				owner ?= {
					has_variable = ztr_un_vote_veto
				}
			}
			custom_tooltip_no_bullet = "ztr_un_vote_map_veto_tt"
		}
	}
	scripted_progress_bar = ztr_un_vote_support_progress_bar
	scripted_progress_bar = ztr_un_vote_neutral_progress_bar
	scripted_progress_bar = ztr_un_vote_against_progress_bar
	scripted_button = ztr_un_vote_for_statement
	scripted_button = ztr_un_vote_against_statement
	scripted_button = ztr_un_vote_veto_statement
	weight = 110
	should_be_pinned_by_default_involved = yes
	can_deactivate = yes
	transferable = no
	progressbar = no
}
