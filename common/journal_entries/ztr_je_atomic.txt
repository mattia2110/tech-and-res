je_develop_nuclear_weapons = {
	icon = "gfx/interface/icons/event_icons/je_atomic_bomb.dds"
	group = je_group_technology
	possible = {
		#Country is actively developing nukes
		has_variable = nuclear_program_started
		#Must be a sovereign state
		is_subject = no
		#Has not already acquired nuclear weapons
		NOT = {
			has_modifier = nuclear_capability
		}
	}
	immediate = {
		set_variable = {
			name = nuclear_development_var
			value = 0
		}
		capital = {
			add_modifier = nuclear_weapon_site
		}
	}
	on_monthly_pulse = {
		effect = {
			nuclear_development_add_progress = yes
		}
	}
	on_yearly_pulse = {
		effect = {
			if = {
				limit = {
					# This will be removed when dev will resolve AI not building great buildings while being in deficit
					any_scope_state = {
						has_modifier = nuclear_weapon_site
						NOR = {
							any_scope_building = {
								is_building_type = building_nuclear_weapons_silo
								is_under_construction = yes
							}
							any_scope_building = {
								is_building_type = building_nuclear_weapons_silo
								is_under_construction = no
							}
						}
					}
					is_ai = yes
					has_technology_researched = atomic_bomb
					ztr_nuclear_research_paused = no
				}
				random_scope_state = {
					limit = {
						has_modifier = nuclear_weapon_site
					}
					create_building = {
						building = building_nuclear_weapons_silo
						level = 1
					}
				}
			}
		}
	}
	complete = {
		scope:journal_entry = {
			is_goal_complete = yes
		}
		#Has nuclear weapons tech
		has_technology_researched = atomic_bomb
		custom_tooltip = {
			text = je_develop_nuclear_weapons_1.tt
			ztr_has_operational_nuclear_silo = yes
		}
		custom_tooltip = {
			text = je_develop_nuclear_weapons_2.tt
			ztr_nuclear_research_paused = no
		}
		has_variable = nuclear_program_started
		has_law_or_variant = law_type:law_uranium_full_usage
		NOT = { has_modifier = nuclear_capability }
		is_subject = no
		is_revolutionary = no
		is_secessionist = no
	}
	on_complete = {
		#Set nuclear capability
		trigger_event = {
			id = nuclear_warfare_system.1
		}
		remove_variable = nuclear_development_var
		remove_variable = nuclear_program_started
		#Add nuclear development cost
		add_modifier = {
			name = nuclear_dev_cost
			years = 5
			is_decaying = yes
		}
	}
	invalid = {
		NAND = {
			#Country is actively developing nukes
			has_variable = nuclear_program_started
			NOT = { has_modifier = nuclear_capability }
			#Has nuclear weapons law
			#has_law_or_variant = law_type:law_wmd_allowed
			has_law_or_variant = law_type:law_uranium_full_usage
			#Must be a sovereign state
			is_subject = no
			is_revolutionary = no
			is_secessionist = no
		}
	}
	on_invalid = {
		ztr_abandon_nuclear_weapons_program = yes
	}
	is_progressing = {
		ztr_nuclear_research_paused = no
	}
	# zero or more static modifiers that will be applied to the Journal Entry when it activates, where they will propagate to the country
	modifiers_while_active = {
		nuclear_dev_cost
	}
	scripted_button = ztr_abandon_nuclear_weapons_program_button
	current_value = {
		value = var:nuclear_development_var
	}
	goal_add_value = {
		value = 1500
	}
	weight = 100
	should_be_pinned_by_default_uninvolved_or_context = yes
	transferable = no
	progressbar = yes
}

je_nuclear_weapons_info_panel = {
	icon = "gfx/interface/icons/event_icons/nuclear_capability.dds"
	group = je_group_foreign_affairs
	is_shown_when_inactive = { has_global_variable = atomic_bomb_discovered }
	possible = { has_global_variable = ztr_atomic_bomb_first_use_var }
	immediate = {
		every_country = {
			limit = { has_variable = first_nuclear_country_var }
			save_scope_as = first_nuclear_country
		}
		if = {
			limit = { NOT = { has_global_variable = ztr_nuclear_scores_initialized } }
			ztr_refresh_nuclear_scores = yes
		}
		ztr_nuclear_ranking_variables = yes
		ztr_nuclear_strength_ranking_effect = yes
		ztr_nuclear_defence_ranking_effect = yes
	}
	weight = 100
	transferable = no
	progressbar = no
}
