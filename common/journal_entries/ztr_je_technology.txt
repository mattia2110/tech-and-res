# National Research Center control panel.
# It has no complete/fail/timeout transition on purpose: it simply activates as soon as the
# country operates a Research Center and deactivates again if that Research Center is lost.
je_research_center = {
	icon = "gfx/interface/icons/event_icons/innovation.dds"
	group = je_group_research_center

	activation_description = {
		always = yes
	}
	is_shown_when_inactive = {
		has_research_center = yes
	}
	possible = {
		has_research_center = yes
	}
	can_deactivate = yes
	# First philosophy; kept if the JE reactivates
	immediate = {
		if = {
			limit = {
				NOT = { has_variable = ztr_rc_philosophy }
			}
			ztr_rc_set_default_philosophy = yes
		}
	}
	# Philosophy modifiers scaled by country_research_progress, only the active one is non-zero
	scaling_modifiers_while_active = {
		modifier = ztr_research_center_independent_modifier
		value = ztr_rc_independent_scale
	}
	scaling_modifiers_while_active = {
		modifier = ztr_research_center_absorption_modifier
		value = ztr_rc_absorption_scale
	}
	scripted_button = ztr_start_nuclear_weapons_program_button
	scripted_button = ztr_develop_thermonuclear_weapons_button
	scripted_button = ztr_nuclear_power_program_button
	scripted_button = ztr_fusion_reactor_button
	scripted_button = ztr_rc_change_philosophy_button
	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_research_center_status_independent
				trigger = { ztr_rc_has_independent_philosophy = yes }
			}
			triggered_desc = {
				desc = je_research_center_status_absorption
				trigger = { ztr_rc_has_absorption_philosophy = yes }
			}
		}
	}
	weight = 50
	should_be_pinned_by_default_uninvolved_or_context = yes
	# Map: owner's weekly innovation against the yearly world leader, white (< 1/8) to scientific blue (>= 7/8)
	map_color = {
		if = {
			limit = {
				ztr_mm_innovation_eighth_1 > 0
				exists = owner
			}
			if = {
				limit = {
					owner = { country_innovation >= ztr_mm_innovation_eighth_7 }
				}
				value = "ztr_innovation_level_7"
			}
			else_if = {
				limit = {
					owner = { country_innovation >= ztr_mm_innovation_eighth_6 }
				}
				value = "ztr_innovation_level_6"
			}
			else_if = {
				limit = {
					owner = { country_innovation >= ztr_mm_innovation_eighth_5 }
				}
				value = "ztr_innovation_level_5"
			}
			else_if = {
				limit = {
					owner = { country_innovation >= ztr_mm_innovation_eighth_4 }
				}
				value = "ztr_innovation_level_4"
			}
			else_if = {
				limit = {
					owner = { country_innovation >= ztr_mm_innovation_eighth_3 }
				}
				value = "ztr_innovation_level_3"
			}
			else_if = {
				limit = {
					owner = { country_innovation >= ztr_mm_innovation_eighth_2 }
				}
				value = "ztr_innovation_level_2"
			}
			else_if = {
				limit = {
					owner = { country_innovation >= ztr_mm_innovation_eighth_1 }
				}
				value = "ztr_innovation_level_1"
			}
			else = {
				value = "ztr_innovation_level_0"
			}
		}
	}
	map_tooltip = {
		if = {
			limit = {
				ztr_mm_innovation_eighth_1 > 0
				exists = owner
			}
			custom_tooltip_no_bullet = "ztr_innovation_map_tt"
		}
	}
}

je_research_center_site = {
	icon = "gfx/interface/icons/event_icons/innovation.dds"
	group = je_group_research_center

	activation_description = {
		always = yes
	}
	on_monthly_pulse = {
		effect = {
			change_variable = {
				name = research_center_var
				add = 1
			}
		}
	}
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			scope:journal_entry = {
				is_goal_complete = yes
			}
			# There should be a state like that, despite research center will always be built in capital
			any_scope_state = {
				ztr_state_can_host_research_center = yes
			}
		}
		effect = {
			#remove_modifier = research_center_survey
			trigger_event = {
				id = ztr_technology.1
			}
			# site found
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			NOT = {
				any_scope_state = {
					ztr_state_can_host_research_center = yes
				}
			}
		}
		effect = {
			remove_modifier = research_center_survey
			remove_variable = research_center_var
		}
	}
	current_value = {
		value = root.var:research_center_var
	}
	goal_add_value = {
		value = 24
	}
	progressbar = yes
	weight = 10
	should_be_pinned_by_default_uninvolved_or_context = yes
}

		trigger = {
			scope:target = {
				has_research_center_target = yes
			}
		}
		effect = {
			trigger_event = {
				id = ztr_technology.2
			}
			remove_variable = research_center_var
			scope:target = {
				remove_modifier = research_center_site
			}
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			NOT = {
				any_scope_state = {
					has_modifier = research_center_site
				}
			}
		}
		effect = {
			remove_variable = research_center_var
		}
	}
	on_yearly_pulse = {
		effect = {
			if = {
				limit = {
					# This will be removed when dev will resolve AI not building great buildings while being in deficit
					capital = {
						NOR = {
							any_scope_building = {
								is_building_type = building_research_center
								is_under_construction = no
							}
							any_scope_building = {
								is_building_type = building_research_center
								is_under_construction = yes
							}
						}
					}
					is_ai = yes
				}
				capital = {
					create_building = {
						building = building_research_center
						level = 1
					}
				}
			}
		}
	}
	should_be_pinned_by_default_uninvolved_or_context = yes
}

# National Electricity System Journal Entries
je_national_electric_system = {
	icon = "gfx/interface/icons/goods_icons/electrical_generation.dds"
	group = je_group_technology

	activation_description = {
		always = yes
	}

	possible = {
		#JE
		has_variable = national_electric_system_started
	}

	immediate = {
		ztr_update_national_electric_system_progress = yes
	}

	on_monthly_pulse = {
		effect = {
			ztr_update_national_electric_system_progress = yes
		}
	}

	on_yearly_pulse = {
		effect = {
			if = {
				limit = {
					is_ai = yes
				}
				every_scope_state = {
					limit = {
						NOR = {
							has_building = building_power_grid_station
							any_scope_building = {
								is_building_type = building_power_grid_station
								is_under_construction = yes
							}
						}
					}
					create_building = {
						building = building_power_grid_station
						level = 1
					}
				}
			}
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			any_scope_state = {
				filter = {
					is_incorporated = yes
				}
				custom_tooltip = {
					text = je_national_electric_system_complete_tt
					has_building = building_power_grid_station
				}
				percent = 1
			}
		}
		effect = {
			ztr_update_national_electric_system_progress = yes
			add_modifier = {
				name = national_electric_system_transition_boost
				years = 10
				is_decaying = yes
			}
			activate_law = law_type:law_national_electric_system
			ztr_clear_national_electric_system_progress = yes
			remove_variable = national_electric_system_started
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_national_electric_system_status
				trigger = {
					always = yes
				}
			}
		}
	}

	current_value = {
		value = var:ztr_nes_states_with_grid_var
	}
	goal_add_value = {
		value = var:ztr_nes_states_total_var
	}

	weight = 100
	should_be_pinned_by_default_uninvolved_or_context = yes
	transferable = no
	progressbar = yes
}

# Fission nuclear power programme, opened by ztr_nuclear_power_program_button.
# nuclear_reactor is can_research = no, so ztr_technology.20 granting it on completion is
# one of the only two ways into the fission branch; the other is buying the patent.
je_nuclear_power_program = {
	icon = "gfx/interface/icons/event_icons/nuclear_reactor_je.dds"
	group = je_group_research_center

	activation_description = {
		always = yes
	}
	possible = {
		has_variable = ztr_nuclear_power_program_started
	}
	immediate = {
		set_variable = {
			name = ztr_nuclear_power_var
			value = 0
		}
	}
	on_monthly_pulse = {
		effect = {
			nuclear_power_add_progress = yes
		}
	}
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			scope:journal_entry = {
				is_goal_complete = yes
			}
		}
		effect = {
			trigger_event = {
				id = ztr_technology.20
			}
			remove_variable = ztr_nuclear_power_var
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			OR = {
				NOT = {
					has_research_center = yes
				}
				has_technology_researched = nuclear_reactor
				# Abandoned from the journal entry's own button
				NOT = {
					has_variable = ztr_nuclear_power_program_started
				}
			}
		}
		effect = {
			remove_variable = ztr_nuclear_power_var
			remove_variable = ztr_nuclear_power_program_started
		}
	}
	# The programme dies with the Research Center, and is pointless if the patent was
	# bought in the meantime.
	scripted_button = ztr_abandon_nuclear_power_program_button
	current_value = {
		value = var:ztr_nuclear_power_var
	}
	goal_add_value = {
		value = 2000
	}
	weight = 100
	should_be_pinned_by_default_uninvolved_or_context = yes
	transferable = no
	progressbar = yes
}
