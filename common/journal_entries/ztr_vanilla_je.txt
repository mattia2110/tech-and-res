REPLACE:je_strike = {
	icon = "gfx/interface/icons/event_icons/event_protest.dds"

	group = je_group_crises

	activation_description = {
		always = yes
	}

	is_shown_when_inactive = {
		ztr_strike_crisis_enabled_trigger = yes #T&R
		has_technology_researched = political_agitation
		has_technology_researched = human_rights
		NOT = { has_law_or_variant = law_type:law_council_republic }
		ig:ig_trade_unions ?= {
			ig_counts_as_marginal = no
			ig_clout >= 0.1
		}
		ig:ig_industrialists ?= {
			ig_counts_as_marginal = no
		}
	}

	possible = {
		ztr_strike_crisis_enabled_trigger = yes #T&R
		ig:ig_trade_unions ?= {
			is_in_government = no
			ig_approval < 0
		}
		OR = {
			institution_investment_level = {
				institution = institution_workplace_safety
				value < 3
			}
			institution_investment_level = {
				institution = institution_social_security
				value < 3
			}
		}
		# we should probably simplfy this
		any_scope_state = {
			is_incorporated = yes
			OR = {
				radical_fraction = {
					value >= 0.2
					pop_type = laborers
				}
				radical_fraction = {
					value >= 0.2
					pop_type = machinists
				}
			}
			any_scope_pop = {
				OR = {
					is_pop_type = laborers
					is_pop_type = machinists
				}
				standard_of_living < 15
				pop_acceptance >= acceptance_status_4
				OR = {
					pop_employment_building_group = bg_manufacturing
					pop_employment_building_group = bg_mining
					pop_employment_building_group = bg_private_infrastructure
				}
			}
		}
	}

	immediate = {
		trigger_event = { id = strike.1 }
		set_variable = {
			name = strike_timer_var
			value = 0
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			strike_goal_complete_trigger = yes
		}
		effect = {
			custom_tooltip = je_end_strike_tt
			if = {
				limit = {
					has_variable = strike_negotiator
				}
				trigger_event = {
					id = strike.8 # strikers appeased
				}
			}
			else_if = {
				limit = {
					has_variable = strike_breaker
				}
				trigger_event = {
					id = strike.9 # strikers crushed
				}
			}
			je_strike_cleanup = yes
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			has_law_or_variant = law_type:law_council_republic
		}
		effect = {
			custom_tooltip = {
				text = end_strike_tt
				strike_end_effect = yes
			}
			je_strike_cleanup = yes
		}
	}

	#T&R: a strike lasting 10 years hardens the proletariat
	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = je_strike_not_ending_after_10_years_tt
				var:strike_timer_var >= 120
				has_variable = liberty_index_var
			}
		}
		effect = {
			trigger_event = { id = ztr_proletariat.1 }
			je_strike_cleanup = yes
		}
	}

	on_monthly_pulse = {
		effect = {
			change_variable = {
				name = strike_timer_var
				add = 1
			}
		}
		events = {
			strike.3
		}
		random_events = {
			10 = 0
			1 = strike.5
			1 = strike.6
			1 = strike.7
		}
	}

	current_value = {
		value = root.var:strike_timer_var
	}

	goal_add_value = {
		value = 60
	}

	progressbar = yes

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_general_strike_warning
				trigger = {
					scope:journal_entry = {
						is_goal_complete = no
					}
				}
			}
			triggered_desc = {
				desc = je_general_strike_active
				trigger = {
					scope:journal_entry = {
						is_goal_complete = yes
					}
				}
			}
		}
	}

	weight = 100
		
	should_be_pinned_by_default_uninvolved_or_context = yes
}

REPLACE:je_warlord_china = {
	icon = "gfx/interface/icons/event_icons/event_fire.dds"

	group = je_group_qing

	activation_description = {
		always = yes
	}

	is_shown_in_lobby = {
        OR = {
			country_has_primary_culture = cu:han
			country_has_primary_culture = cu:manchu
		}
    }

	immediate = {
		random_law = {
			limit = { type = law_type:law_monarchy }
			save_scope_as = monarchy_scope
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			OR = { #T&R: the warlord era cannot outlast 1940
				AND = {
					custom_tooltip = {
						text = won_boxer_tt
						has_variable = won_boxer_rebellion
					}
					custom_tooltip = {
						text = china_recognized_tt
						has_variable = china_warlord_explosion_recognized
					}
					custom_tooltip = {
						text = china_abolished_monarchy_tt
						has_variable = china_warlord_abolished_monarchy
					}
				}
				year > 1940
			}
		}
		effect = {
			if = { #T&R
				limit = {
					year <= 1940
				}
				trigger_event = { id = warlord_china_events.200 }
			}
			else = {
				trigger_event = { id = the_future_of_china.201 }
			}
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = je_china_failure_tt
				var:china_warlord_explosion <= 0
			}
		}
		effect = {
			trigger_event = { id = warlord_china_events.100 }
		}
	}

	current_value = {
		value = var:china_warlord_explosion
	}

	goal_add_value = {
		add = 5
	}

	progressbar = yes

	on_weekly_pulse = {
		effect = {
			if = {
				limit = {
					NOT = { has_variable = china_warlord_explosion_subject }
					is_subject = yes
				}
				chi_remove_1_fragile_unity = yes
				set_variable = china_warlord_explosion_subject
			}
			if = {
				limit = {
					NOT = { has_variable = china_warlord_explosion_loans }
					taking_loans = yes
				}
				chi_remove_1_fragile_unity = yes
				set_variable = china_warlord_explosion_loans
			}
			if = {
				limit = {
					NOT = { has_variable = china_warlord_explosion_recognized }
					country_rank != rank_value:unrecognized_power
					country_rank != rank_value:unrecognized_major_power
					country_rank != rank_value:unrecognized_regional_power
				}
				set_variable = china_warlord_explosion_recognized
				chi_add_1_fragile_unity = yes
			}
			if = {
				limit = {
					NOT = { has_variable = china_abolished_monarchy }
					NOT = { country_has_monarchy_law = yes }
				}
				set_variable = china_abolished_monarchy
				chi_remove_1_fragile_unity = yes
				if = {
					limit = {
						NOT = { has_variable = china_han_empire }
					}
					trigger_event = { id = boxer_rebellion_events.4 days = 0 popup = yes }
				}
			}
		}
	}

	weight = 10000
		
	should_be_pinned_by_default_uninvolved_or_context = yes
}

REPLACE:je_social_revolution = {
	icon = "gfx/interface/icons/generic_icons/golden_rose.dds"

	group = je_group_socialism

	activation_description = {
		always = yes
	}

	immediate = {
		ig:ig_trade_unions ?= {
			save_scope_as = proletariat_ig
		}
		set_variable = { name = communist_welfare_low_consolidation_months_var value = 0 }
	}

	on_monthly_pulse = {
		effect = {
			if = {
				limit = {
					has_journal_entry = je_state_and_revolution
				}
				if = {
					limit = {
						AND = {
							"je:je_state_and_revolution.scripted_bar_progress(je_state_and_revolution_political_bar)" < 50
							"je:je_state_and_revolution.scripted_bar_progress(je_state_and_revolution_economic_bar)" < 50
						}
					}
					change_variable = { name = communist_welfare_low_consolidation_months_var add = 1 }
				}
				else_if = {
					limit = {
						var:communist_welfare_low_consolidation_months_var > 0
					}
					change_variable = { name = communist_welfare_low_consolidation_months_var subtract = 1 }
				}
			}
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			has_law_or_variant = law_type:law_multicultural
			has_law_or_variant = law_type:law_womens_suffrage
			has_law_or_variant = law_type:law_no_colonial_affairs
			AND = {
				has_law_or_variant = law_type:law_public_schools
				institution_investment_level = {
					institution = institution_schools
					value = 5
				}
			}
			# T&R: the 5th Health System level comes in era 8, so level 4 plus the tech that gives it in vanilla
			AND = {
				has_law_or_variant = law_type:law_public_health_insurance
				institution_investment_level = {
					institution = institution_health_system
					value >= 4
				}
				has_technology_researched = malaria_prevention
			}
			# T&R: the 5th Social Security level comes in era 6, so level 4 plus the tech that gives it in vanilla
			AND = {
				has_law_or_variant = law_type:law_old_age_pension
				institution_investment_level = {
					institution = institution_social_security
					value >= 4
				}
				has_technology_researched = feminism
			}
			AND = {
				has_law_or_variant = law_type:law_worker_protections
				institution_investment_level = {
					institution = institution_workplace_safety
					value = 5
				}
			}
			approaching_bureaucracy_shortage = no
		}
		effect = {
			trigger_event = {
				id = communism.6
				popup = yes
			}
		}
		event_outcome_effect_desc = {
			effect = {
				add_modifier = {
					name = modifier_social_revolution
					days = very_long_modifier_time
				}
			}
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			OR = {
				NOT = { has_law_or_variant = law_type:law_council_republic }
				custom_tooltip = {
					text = je_social_revolution_low_consolidation_tt
					var:communist_welfare_low_consolidation_months_var >= 120
				}
				custom_tooltip = {
					text = je_social_revolution_concluded_consolidation_tt
					has_modifier = modifier_state_and_revolution_conciliation
				}
			}
		}
		effect = {
			ig:ig_armed_forces ?= {
				if = {
					limit = {
						has_ideology = ideology:ideology_internationalist
					}
					add_ideology = ideology_defensist
				}
			}
			every_interest_group = {
				limit = {
					has_ideology = ideology:ideology_internationalist
				}
				remove_ideology = ideology_internationalist
			}
			je_social_revolution_cleanup = yes
		}
	}

	should_be_pinned_by_default_uninvolved_or_context = yes
}
