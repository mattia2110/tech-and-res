# =====================================================================================
# SUPERSTATE FLAGS SELECTION
# Without these definitions the game would always fall back to the coat of arms named
# after the tag, and the "_communist" / "_fascist" variants in
# ztr_superstates_countries.txt would never be shown.
#
# The trigger reads the same ztr_superstate_ideology country variable used by
# ztr_superstates_dynamic_names.txt (2 = commune, 3 = imperium). When the variable is
# unset no definition applies and the base flag is used.
# =====================================================================================

EUF = { # European Federation
	flag_definition = {
		coa = EUF_communist
		subject_canton = EUF_communist
		priority = 100
		trigger = {
			exists = scope:actor
			scope:actor ?= {
				has_variable = ztr_superstate_ideology
				var:ztr_superstate_ideology = 2
			}
		}
	}
	flag_definition = {
		coa = EUF_fascist
		subject_canton = EUF_fascist
		priority = 100
		trigger = {
			exists = scope:actor
			scope:actor ?= {
				has_variable = ztr_superstate_ideology
				var:ztr_superstate_ideology = 3
			}
		}
	}
}
