# Social conditions read by the demographic extras in ztr_demography_values.txt

# Society
ztr_demographic_has_healthcare = {
	OR = {
		has_law_or_variant = law_type:law_private_health_insurance
		has_law_or_variant = law_type:law_public_health_insurance
	}
}

ztr_demographic_slavery_legal = {
	NOT = {
		has_law_or_variant = law_type:law_slavery_banned
	}
}

ztr_demographic_traditional_society = {
	OR = {
		country_has_state_religion = rel:sunni
		country_has_state_religion = rel:shiite
		c:DEI ?= this
		c:BIC ?= this
	}
}

# Open borders only matter where there is somewhere to migrate from
ztr_demographic_open_to_immigration = {
	OR = {
		has_law_or_variant = law_type:law_multicultural
		has_law_or_variant = law_type:law_no_migration_controls
	}
	OR = {
		total_population < 200000000
		is_in_geographic_region = geographic_region_americas
	}
}
