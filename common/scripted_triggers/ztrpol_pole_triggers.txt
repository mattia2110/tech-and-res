# Tech&Res politics: Liberty/Equality poles (scope: country), set yearly by ztrpol_update_pole

# High liberty, high equality
ztrpol_is_social_pole = {
	custom_tooltip = {
		text = ztrpol_tooltip_is_social_pole
		has_variable = ztrpol_pole_set
		has_variable = ztrpol_high_liberty
		has_variable = ztrpol_high_equality
	}
}

# High liberty, low equality
ztrpol_is_liberal_pole = {
	custom_tooltip = {
		text = ztrpol_tooltip_is_liberal_pole
		has_variable = ztrpol_pole_set
		has_variable = ztrpol_high_liberty
		NOT = {
			has_variable = ztrpol_high_equality
		}
	}
}

# Low liberty, high equality
ztrpol_is_collectivist_pole = {
	custom_tooltip = {
		text = ztrpol_tooltip_is_collectivist_pole
		has_variable = ztrpol_pole_set
		NOT = {
			has_variable = ztrpol_high_liberty
		}
		has_variable = ztrpol_high_equality
	}
}

# Low liberty, low equality
ztrpol_is_reactionary_pole = {
	custom_tooltip = {
		text = ztrpol_tooltip_is_reactionary_pole
		has_variable = ztrpol_pole_set
		NOT = {
			has_variable = ztrpol_high_liberty
		}
		NOT = {
			has_variable = ztrpol_high_equality
		}
	}
}

# Pole cores: liberty 20 points past the frontier at 50 (democratic >= 70, authoritarian < 30)
ztrpol_is_social_core = {
	custom_tooltip = {
		text = ztrpol_tooltip_is_social_core
		ztrpol_is_social_pole = yes
		var:liberty_index_var >= 70
	}
}

ztrpol_is_liberal_core = {
	custom_tooltip = {
		text = ztrpol_tooltip_is_liberal_core
		ztrpol_is_liberal_pole = yes
		var:liberty_index_var >= 70
	}
}

ztrpol_is_collectivist_core = {
	custom_tooltip = {
		text = ztrpol_tooltip_is_collectivist_core
		ztrpol_is_collectivist_pole = yes
		var:liberty_index_var < 30
	}
}

ztrpol_is_reactionary_core = {
	custom_tooltip = {
		text = ztrpol_tooltip_is_reactionary_core
		ztrpol_is_reactionary_pole = yes
		var:liberty_index_var < 30
	}
}

# The pole differs from the stored code (or has never been stored)
ztrpol_pole_changed = {
	OR = {
		NOT = {
			has_variable = ztrpol_pole_code
		}
		AND = {
			ztrpol_is_social_pole = yes
			NOT = {
				var:ztrpol_pole_code = 1
			}
		}
		AND = {
			ztrpol_is_liberal_pole = yes
			NOT = {
				var:ztrpol_pole_code = 2
			}
		}
		AND = {
			ztrpol_is_collectivist_pole = yes
			NOT = {
				var:ztrpol_pole_code = 3
			}
		}
		AND = {
			ztrpol_is_reactionary_pole = yes
			NOT = {
				var:ztrpol_pole_code = 4
			}
		}
	}
}
