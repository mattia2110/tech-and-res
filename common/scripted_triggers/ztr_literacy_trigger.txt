
country_has_academic_low_cap = {
	OR = {
		literacy_rate > 0.5
		country_has_academic_base_level_1 = yes
	}
}

country_has_academic_medium_cap = {
	country_has_academic_base_level_1 = yes
	OR = {
		literacy_rate > 0.8
		country_has_academic_base_level_3 = yes
	}
}

country_has_academic_high_cap = {
	country_has_academic_base_level_3 = yes
	OR = {
		literacy_rate > 0.95
		country_has_academic_base_level_4 = yes
	}
}
