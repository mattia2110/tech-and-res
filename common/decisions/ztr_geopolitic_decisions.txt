# Found or join the European Economic Community
firm_ztr_treaty_of_rome = {
	is_shown = {
		has_technology_researched = socialdemocracy
		year > 1950
		is_subject = no
		ztr_is_ecc_country = yes
		is_not_isolationist_switzerland = yes
		NOT = {
			has_modifier = ztr_eec_member
		}
		NOT = {
			has_modifier = ztr_eu_member
		}
		# Once the EU exists the EEC chapter is closed: new countries join the Union directly
		NOT = {
			has_global_variable = ztr_eu_created
		}
	}
	possible = {
		ztr_is_un_member = yes
		is_at_war = no
		is_revolutionary = no
	}
	when_taken = {
		change_infamy = -5
		hidden_effect = {
			if = {
				limit = {
					NOT = {
						has_global_variable = ztr_eec_created
					}
				}
				set_global_variable = ztr_eec_created
				set_global_variable = {
					name = ztr_eu_phase
					value = 1
				}
			}
		}
		add_modifier = {
			name = ztr_eec_member
		}
		every_country = {
			limit = {
				NOT = { this = root }
				has_modifier = ztr_eec_member
			}
			change_relations = {
				country = root
				value = 200
			}
		}
		every_rival_country = {
			limit = {
				has_modifier = ztr_eec_member
			}
			remove_diplomatic_pact = {
				country = root
				type = rivalry
			}
			change_relations = {
				country = root
				value = 200
			}
		}
	}
	ai_chance = {
		value = 25
	}
}

# Join the existing European Union
join_european_union = {
	is_shown = {
		has_technology_researched = transnational_activism
		has_global_variable = ztr_eu_created
		global_var:ztr_eu_phase < 4
		is_subject = no
		ztr_is_european_country = yes
		NOT = {
			has_modifier = ztr_eu_member
		}
		NOT = {
			has_variable = ztr_eu_rejoin_cd
		}
	}
	possible = {
		ztr_is_un_member = yes
		is_at_war = no
		is_revolutionary = no
		NOT = { infamy >= 25 }
		# Copenhagen-style criteria: the Union only opens to politically aligned countries
		ztr_eu_ideology_compatible = yes
		is_not_isolationist_switzerland = yes
		custom_tooltip = {
			text = ztr_eu_join_chair_relations_tt
			any_country = {
				has_variable = ztr_eu_founder
				"relations(root)" >= relations_threshold:cordial
			}
		}
	}
	when_taken = {
		# If a Country joins the EU, all of its European subjects will become independent.
		# Overseas dependencies are kept: the EEC founders joined while holding colonial empires.
		custom_tooltip = {
			text = ztr_eu_join_country_loose_all_subjects_tt
			every_subject_or_below = {
				limit = { ztr_is_european_country = yes }
				make_independent = yes
			}
		}
		change_infamy = -5
		ztr_eu_member_accession = yes
	}
	ai_chance = {
		value = 50
	}
}

# Withdraw from the European Union
withdraw_from_european_union = {
	is_shown = {
		has_modifier = ztr_eu_member
		year > 2010 #TODO Da sostituire con righ-wing populism tech
		is_subject = no
		NOT = {
			has_variable = ztr_eu_founder
		}
	}
	possible = {
		custom_tooltip = {
			text = ztr_eu_vote_session_clear_tt
			NOT = {
				has_global_variable = ztr_eu_vote_active
			}
		}
		custom_tooltip = {
			text = ztr_eu_withdraw_euroscepticism_tt
			OR = {
				is_ai = no
				ig:ig_petty_bourgeoisie = {
					is_powerful = yes
				}
				NOT = {
					ztr_eu_ideology_compatible = yes
				}
			}
		}
	}
	when_taken = {
		change_infamy = 5
		ztr_eu_member_exit = yes
	}
	ai_chance = {
		value = 5
	}
}

# Join the sovereign European Federation after the EU institutions have dissolved
join_european_federation = {
	is_shown = {
		has_global_variable = ztr_eu_federation_formed
		exists = c:EUF
		NOT = { this = c:EUF }
		is_subject = no
		ztr_is_european_country = yes
		is_not_isolationist_switzerland = yes
		NOT = {
			has_variable = ztr_eu_rejoin_cd
		}
	}
	possible = {
		is_at_war = no
		is_revolutionary = no
		NOT = { infamy >= 25 }
		ztr_euf_ideology_compatible = yes
	}
	when_taken = {
		# As with EU accession, European dependencies are freed; overseas ones are kept
		custom_tooltip = {
			text = ztr_eu_join_country_loose_all_subjects_tt
			every_subject_or_below = {
				limit = { ztr_is_european_country = yes }
				make_independent = yes
			}
		}
		ztr_eu_join_existing_federation = yes
	}
	ai_chance = {
		value = 25
	}
}

# Adopt the Euro (open from the EU phase onwards, convergence criteria apply)
ztr_adopt_euro_decision = {
	is_shown = {
		has_modifier = ztr_eu_member
		has_global_variable = ztr_eu_created
		is_not_isolationist_switzerland = yes
		has_technology_researched = electronic_payment_systems
		global_var:ztr_eu_phase >= 2
		global_var:ztr_eu_phase < 4
		NOT = {
			has_modifier = ztr_euro_member
		}
	}
	possible = {
		ztr_eu_euro_convergence_criteria = yes
	}
	when_taken = {
		ztr_eu_adopt_euro = yes
	}
	ai_chance = {
		# The AI adopts the Euro through the Union panel once it qualifies
		value = 0
	}
}
