start_national_electric_system_conversion_decision = {
	is_shown = {
		has_technology_researched = national_power_grid
		has_law_or_variant = law_type:law_no_national_electric_system
		NOT = {
			has_journal_entry = je_national_electric_system
		}
	}
	possible = {
		always = yes
	}
	when_taken = {
		set_variable = national_electric_system_started
		add_journal_entry = {
			type = je_national_electric_system
		}
	}
	ai_chance = {
		value = 0
		if = {
			limit = {
				#is_at_war = no
				scaled_debt < 0.25
			}
			add = 1000
		}
	}
}

# License first generation reactor technology from a country that already operates one.
nuclear_reactor_purchase_decision = {
	is_shown = {
		# Research Center or the universities linkage: has_research_modernized covers both.
		has_research_modernized = yes
		has_technology_researched = national_power_grid
		NOT = {
			has_technology_researched = nuclear_reactor
		}
		NOT = {
			has_variable = nuclear_reactor_purchase_cooldown
		}
	}
	possible = {
		produced_bureaucracy > 500
		# Literacy is high or number of literate pop is high
		country_has_academic_low_cap = yes
		# Someone has to be willing to sell
		custom_tooltip = {
			text = nuclear_reactor_purchase_relations_tt
			any_country = {
				NOT = {
					this = root
				}
				has_technology_researched = nuclear_reactor
				OR = {
					relations:root >= relations_threshold:cordial
					root = {
						is_subject_of = PREV
					}
					is_in_same_power_bloc = root
				}
			}
		}
	}
	when_taken = {
		# Set 3-year cooldown
		set_variable = {
			name = nuclear_reactor_purchase_cooldown
			years = 3
		}
		save_scope_as = nuclear_technology_requester
		random_country = {
			limit = {
				NOT = {
					this = root
				}
				has_technology_researched = nuclear_reactor
				OR = {
					relations:root >= relations_threshold:cordial
					root = {
						is_subject_of = PREV
					}
					is_in_same_power_bloc = root
				}
			}
			save_scope_as = nuclear_technology_seller
			trigger_event = {
				id = ztr_technology.21
				popup = yes
			}
		}
	}
	ai_chance = {
		value = 0
		if = {
			limit = {
				produced_bureaucracy > 2000
			}
			add = 500
		}
		if = {
			limit = {
				has_strategy = ai_strategy_industrial_expansion
			}
			add = 5000
		}
		if = {
			limit = {
				OR = {
					is_diplomatic_play_committed_participant = yes
					is_at_war = yes
				}
			}
			multiply = 0
		}
		# Do not buy what we can build ourselves.
		if = {
			limit = {
				has_research_center = yes
				country_research_progress >= 22
				NOT = {
					any_country = {
						NOT = {
							this = root
						}
						has_technology_researched = nuclear_reactor
						OR = {
							relations:root >= relations_threshold:cordial
							root = {
								is_subject_of = PREV
							}
							is_in_same_power_bloc = root
						}
						prestige >= root.prestige
					}
				}
			}
			multiply = 0
		}
	}
}
