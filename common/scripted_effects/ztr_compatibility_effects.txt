# REVOLUTION IG ROLES
#############################

# Hook: Russian soviet factions, Trade Unions and Rural Folk in vanilla
ztr_russia_save_soviet_ig_scopes = {
	ig:ig_trade_unions ?= {
		save_scope_as = russia_bolsheviks_scope
		save_scope_as = russia_mensheviks_scope
		save_scope_as = russia_soviet_winner_scope
	}
	ig:ig_rural_folk ?= {
		save_scope_as = russia_left_sr_scope
	}
}

# Hook: further ruling IGs of the agro-socialist soviet republic
ztr_russia_agro_socialist_coalition = {}

# Hook: further ruling IGs of communist China and the IG providing the ruler
ztr_china_communist_party_takes_power = {
	ig:ig_trade_unions ?= {
		leader = {
			set_character_as_ruler = yes
		}
	}
}
