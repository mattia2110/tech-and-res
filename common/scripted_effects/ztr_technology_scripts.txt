# Refreshes the Research Center research output modifier. Scope: country
ztr_update_research_center_progress = {
	if = {
		limit = {
			has_modifier = ztr_research_center_progress_modifier
		}
		remove_modifier = ztr_research_center_progress_modifier
	}
	if = {
		limit = {
			country_research_progress > 0
		}
		add_modifier = {
			name = ztr_research_center_progress_modifier
			multiplier = country_research_progress
			years = 2
		}
	}
}

# Monthly progress for je_fusion_reactor_research. Scope: country
fusion_reactor_add_progress = {
	# One point of progress per point of country_research_progress.
	change_variable = {
		name = fusion_reactor_var
		add = {
			value = country_research_progress
			subtract = 14
			min = 1
		}
	}
}

# Monthly progress for je_develop_nuclear_weapons.
nuclear_development_add_progress = {
	change_variable = {
		name = nuclear_development_var
		add = {
			value = country_research_progress
			subtract = 5
			min = 1
			# The need to rush if rivals have nukes or if at war
			if = {
				limit = {
					OR = {
						any_rival_country = {
							has_modifier = nuclear_capability
						}
						is_at_war = yes
					}
				}
				multiply = 2
			}
			# Diplomatic and operational pauses preserve accumulated progress.
			if = {
				limit = {
					ztr_nuclear_research_paused = yes
				}
				multiply = 0
			}
		}
	}
}

# Monthly progress for je_develop_thermonuclear_weapons.
thermonuclear_development_add_progress = {

# Monthly progress for je_nuclear_power_program. Scope: country
# The button asks for the atomic_bomb technology, not for warheads. A country that has
# actually built and tested them already runs the enrichment plants, the production
# reactors and the engineers, so it gets to civil power twice as fast.
nuclear_power_add_progress = {
	change_variable = {
		name = ztr_nuclear_power_var
		add = {
			value = country_research_progress
			subtract = 5
			min = 1
			if = {
				limit = {
					has_modifier = nuclear_capability
				}
				multiply = 2
			}
		}
	}
}

# Abandonment keeps the eight year launch cooldown and diplomatic consequences.
ztr_abandon_nuclear_weapons_program = {
	remove_variable = nuclear_program_started
	remove_variable = nuclear_development_var
	if = {
		limit = {
			NOT = { has_modifier = nuclear_capability }
		}
		every_scope_state = {
			limit = { has_modifier = nuclear_weapon_site }
			remove_modifier = nuclear_weapon_site
		}
	}
}

# Abandons the thermonuclear programme. The infamy and relation hits taken when it
# was launched are not reverted.
ztr_abandon_thermonuclear_weapons_program = {
	remove_variable = thermonuclear_program_started
	remove_variable = thermonuclear_development_var
}

# Abandons the civil nuclear power programme. Nothing else was spent to start it.
ztr_abandon_nuclear_power_program = {
	remove_variable = ztr_nuclear_power_program_started
	remove_variable = ztr_nuclear_power_var
}

# Abandons the fusion programme. fusion_reactor_var doubles as the started marker
# for this one, so removing it is the whole reset.
ztr_abandon_fusion_reactor_program = {
	remove_variable = fusion_reactor_var
}

# Shared acquisition path for a completed test or a recovered tested arsenal.
ztr_grant_nuclear_capability = {
	add_modifier = nuclear_capability
	remove_variable = nuclear_program_started
	remove_variable = nuclear_development_var
	save_scope_as = nuclear_country
	every_scope_state = {
		limit = { has_building = building_nuclear_weapons_silo }
		set_variable = ztr_tested_nuclear_arsenal
		add_modifier = nuclear_weapon_site
	}
	if = {
		limit = { has_technology_researched = hydrogen_bomb }
		set_variable = thermonuclear_capability_var
	}
	calculate_nuclear_strength_info = yes
	calculate_nuclear_defence_info = yes
	if = {
		limit = { NOT = { has_global_variable = atomic_bomb_discovered } }
		if = {
			limit = {
				NOT = { any_country = { has_variable = first_nuclear_country_var } }
			}
			set_variable = first_nuclear_country_var
		}
		save_scope_as = ztr_first_nuclear_discovery_country
		set_global_variable = atomic_bomb_discovered
		every_country = {
			limit = { NOT = { this = scope:nuclear_country } }
			trigger_event = nuclear_warfare_system.8
		}
	}
}
