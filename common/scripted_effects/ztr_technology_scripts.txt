# First philosophy, without cooldown: independent research from 40 million literates
ztr_rc_set_default_philosophy = {
	if = {
		limit = { country_has_academic_base_level_4 = yes }
		set_variable = { name = ztr_rc_philosophy value = flag:independent }
	}
	else = {
		set_variable = { name = ztr_rc_philosophy value = flag:absorption }
	}
}

# Philosophy chosen from the Research Center panel; $PHILOSOPHY$ = independent or absorption
ztr_rc_change_philosophy = {
	set_variable = { name = ztr_rc_philosophy value = flag:$PHILOSOPHY$ }
	set_variable = {
		name = ztr_rc_philosophy_cooldown
		years = 10
	}
}

# Research agreement entered into force between scope:ztr_ra_first and scope:ztr_ra_second
ztr_research_agreement_signed_effect = {
	ztr_research_agreement_order_partners = yes
	if = {
		limit = { exists = scope:ztr_ra_larger }
		# The larger partner gains a friendly lobby in the smaller one
		add_lobby_appeasement_from_diplomacy_unidirectional = {
			FIRST = scope:ztr_ra_smaller
			SECOND = scope:ztr_ra_larger
			PRO_AMOUNT = 3
			ANTI_AMOUNT = -3
			FACTOR = appeasement_ztr_research_agreement_signed
		}
		scope:ztr_ra_smaller = {
			create_diplomatic_catalyst = {
				type = catalyst_ztr_research_agreement_signed
				target = scope:ztr_ra_larger
			}
		}
	}
	else = {
		add_lobby_appeasement_from_diplomacy_bidirectional = {
			FIRST = scope:ztr_ra_first
			SECOND = scope:ztr_ra_second
			PRO_AMOUNT = 2
			ANTI_AMOUNT = -2
			FACTOR = appeasement_ztr_research_agreement_signed
		}
	}
}

# Research agreement withdrawn: scope:ztr_ra_first withdraws from scope:ztr_ra_second
ztr_research_agreement_ended_effect = {
	ztr_research_agreement_order_partners = yes
	if = {
		limit = { exists = scope:ztr_ra_larger }
		add_lobby_appeasement_from_diplomacy_unidirectional = {
			FIRST = scope:ztr_ra_smaller
			SECOND = scope:ztr_ra_larger
			PRO_AMOUNT = -3
			ANTI_AMOUNT = 3
			FACTOR = appeasement_ztr_research_agreement_ended
		}
	}
	else = {
		add_lobby_appeasement_from_diplomacy_bidirectional = {
			FIRST = scope:ztr_ra_first
			SECOND = scope:ztr_ra_second
			PRO_AMOUNT = -2
			ANTI_AMOUNT = 2
			FACTOR = appeasement_ztr_research_agreement_ended
		}
	}
}

# Saves ztr_ra_larger and ztr_ra_smaller when the two partners differ in rank step
ztr_research_agreement_order_partners = {
	if = {
		limit = {
			scope:ztr_ra_first.ztr_research_rank_step > scope:ztr_ra_second.ztr_research_rank_step
		}
		scope:ztr_ra_first = { save_scope_as = ztr_ra_larger }
		scope:ztr_ra_second = { save_scope_as = ztr_ra_smaller }
	}
	else_if = {
		limit = {
			scope:ztr_ra_second.ztr_research_rank_step > scope:ztr_ra_first.ztr_research_rank_step
		}
		scope:ztr_ra_second = { save_scope_as = ztr_ra_larger }
		scope:ztr_ra_first = { save_scope_as = ztr_ra_smaller }
	}
}

# Research agreement declined (treaty or diplomatic action): -10 relations, pair blocked for 2 years
ztr_research_agreement_declined_effect = {
	scope:ztr_ra_first = {
		change_relations = {
			country = scope:ztr_ra_second
			value = -10
		}
		add_to_variable_list = {
			name = ztr_research_agreement_declined
			target = scope:ztr_ra_second
			years = 2
		}
	}
	scope:ztr_ra_second = {
		add_to_variable_list = {
			name = ztr_research_agreement_declined
			target = scope:ztr_ra_first
			years = 2
		}
	}
}

# Monthly progress for je_fusion_reactor_research. Scope: country
fusion_reactor_add_progress = {
	# One point of progress per point of country_research_progress.
	change_variable = {
		name = fusion_reactor_var
		add = {
			value = country_research_progress
			subtract = 14
			min = 1
		}
	}
}

# Monthly progress for je_develop_nuclear_weapons.
# Progress stops at 1200 (80% of goal_add_value) while any other complete condition is unmet.
nuclear_development_add_progress = {
	if = {
		limit = {
			OR = {
				ztr_nuclear_weapons_goals_met = yes
				var:nuclear_development_var < 1200
			}
		}
		change_variable = {
			name = nuclear_development_var
			add = {
				value = country_research_progress
				subtract = 5
				min = 1
				# The need to rush if rivals have nukes or if at war
				if = {
					limit = {
						OR = {
							any_rival_country = {
								has_modifier = nuclear_capability
							}
							is_at_war = yes
						}
					}
					multiply = 2
				}
				# A third of the pace until the required technology is researched
				if = {
					limit = {
						NOT = { has_technology_researched = atomic_bomb }
					}
					divide = 3
				}
				# Diplomatic and operational pauses preserve accumulated progress.
				if = {
					limit = {
						ztr_nuclear_research_paused = yes
					}
					multiply = 0
				}
			}
		}
		if = {
			limit = {
				ztr_nuclear_weapons_goals_met = no
			}
			clamp_variable = { name = nuclear_development_var max = 1200 min = 0 }
		}
	}
}

# Monthly progress for je_nuclear_power_program. Scope: country
# The button asks for the atomic_bomb technology, not for warheads. A country that has
# actually built and tested them already runs the enrichment plants, the production
# reactors and the engineers, so it gets to civil power twice as fast.
nuclear_power_add_progress = {
	change_variable = {
		name = ztr_nuclear_power_var
		add = {
			value = country_research_progress
			subtract = 5
			min = 1
			if = {
				limit = {
					has_modifier = nuclear_capability
				}
				multiply = 2
			}
		}
	}
}

# Abandonment keeps the eight year launch cooldown and diplomatic consequences.
ztr_abandon_nuclear_weapons_program = {
	remove_variable = nuclear_program_started
	remove_variable = nuclear_development_var
	if = {
		limit = {
			NOT = { has_modifier = nuclear_capability }
		}
		every_scope_state = {
			limit = { has_modifier = nuclear_weapon_site }
			remove_modifier = nuclear_weapon_site
		}
	}
}

# Abandons the thermonuclear programme. The infamy and relation hits taken when it
# was launched are not reverted.
ztr_abandon_thermonuclear_weapons_program = {
	remove_variable = thermonuclear_program_started
	remove_variable = thermonuclear_development_var
}

# Abandons the civil nuclear power programme. Nothing else was spent to start it.
ztr_abandon_nuclear_power_program = {
	remove_variable = ztr_nuclear_power_program_started
	remove_variable = ztr_nuclear_power_var
}

# Abandons the fusion programme. fusion_reactor_var doubles as the started marker
# for this one, so removing it is the whole reset.
ztr_abandon_fusion_reactor_program = {
	remove_variable = fusion_reactor_var
}

# Shared acquisition path for a completed test or a recovered tested arsenal.
ztr_grant_nuclear_capability = {
	add_modifier = nuclear_capability
	remove_variable = nuclear_program_started
	remove_variable = nuclear_development_var
	save_scope_as = nuclear_country
	every_scope_state = {
		limit = { has_building = building_nuclear_weapons_silo }
		set_variable = ztr_tested_nuclear_arsenal
		add_modifier = nuclear_weapon_site
	}
	if = {
		limit = { has_technology_researched = hydrogen_bomb }
		set_variable = thermonuclear_capability_var
	}
	calculate_nuclear_strength_info = yes
	calculate_nuclear_defence_info = yes
	if = {
		limit = { NOT = { has_global_variable = atomic_bomb_discovered } }
		if = {
			limit = {
				NOT = { any_country = { has_variable = first_nuclear_country_var } }
			}
			set_variable = first_nuclear_country_var
		}
		save_scope_as = ztr_first_nuclear_discovery_country
		set_global_variable = atomic_bomb_discovered
		every_country = {
			limit = { NOT = { this = scope:nuclear_country } }
			trigger_event = nuclear_warfare_system.8
		}
	}
}
