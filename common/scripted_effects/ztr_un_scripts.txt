ztr_un_vote_clear_results = {
	remove_global_variable = ztr_un_vote_support_count
	remove_global_variable = ztr_un_vote_neutral_count
	remove_global_variable = ztr_un_vote_against_count
	remove_global_variable = ztr_un_vote_total_count
	remove_global_variable = ztr_un_vote_support_percent
	remove_global_variable = ztr_un_vote_neutral_percent
	remove_global_variable = ztr_un_vote_against_percent
}

join_un_law_changes = {
	if = {
		limit = {
			has_law_or_variant = law_type:law_no_health_system
		}
		activate_law = law_type:law_charitable_health_system
	}
	# The United Nations will abolish outlawed dissent
	if = {
		limit = {
			OR = {
				has_law_or_variant = law_type:law_outlawed_dissent
			}
		}
		activate_law = law_type:law_censorship
	}
	# The United Nations will abolish racist laws if they are still present
	if = {
		limit = {
			OR = {
				has_law_or_variant = law_type:law_ethnostate
				has_law_or_variant = law_type:law_national_supremacy
			}
		}
		activate_law = law_type:law_racial_segregation
	}
	#TODO Modificare questa parte con la modifica forzata delle leggi attive per ogni membro dell'ONU
	# Apply active UN Human Rights resolutions to new members
	# Decolonization Stance
	# Encouraged
	if = {
		limit = {
			has_global_variable = ztr_un_hr_decolonization
			global_var:ztr_un_hr_decolonization = 2
			OR = {
				has_law_or_variant = law_type:law_colonial_resettlement
				has_law_or_variant = law_type:law_colonial_exploitation
				has_law_or_variant = law_type:law_frontier_colonization
			}
		}
		activate_law = law_type:law_no_colonial_affairs
	}
	# Conservative
	else_if = {
		limit = {
			has_global_variable = ztr_un_hr_decolonization
			global_var:ztr_un_hr_decolonization = 1
			OR = {
				has_law_or_variant = law_type:law_colonial_resettlement
				has_law_or_variant = law_type:law_colonial_exploitation
			}
		}
		activate_law = law_type:law_no_colonial_affairs
	}
	# Slavery Stance
	if = {
		limit = {
			has_global_variable = ztr_un_hr_slavery
			global_var:ztr_un_hr_slavery = 1
			OR = {
				has_law_or_variant = law_type:law_colonial_slavery
				has_law_or_variant = law_type:law_debt_slavery
				has_law_or_variant = law_type:law_slave_trade
				has_law_or_variant = law_type:law_legacy_slavery
			}
		}
		activate_law = law_type:law_slavery_banned
		set_variable = {
			name = ztr_recently_banned_slavery
			years = 10
		}
	}
	#Children's Rights
	if = {
		limit = {
			has_global_variable = ztr_un_hr_children_rights
			global_var:ztr_un_hr_children_rights = 1
			has_law_or_variant = law_type:law_child_labor_allowed
		}
		activate_law = law_type:law_restricted_child_labor
	}
	# Compulsory school keeps the restricted child labor floor, then is requested by ztr_un_compliance.2
	else_if = {
		limit = {
			has_global_variable = ztr_un_hr_children_rights
			global_var:ztr_un_hr_children_rights = 2
			has_law_or_variant = law_type:law_child_labor_allowed
		}
		activate_law = law_type:law_restricted_child_labor
	}
	#Women's rights: requested by ztr_un_compliance.3, not imposed
	# Environment and labor rights minimum standards
	ztr_un_apply_environment_standard = yes
	ztr_un_apply_labor_rights_standard = yes
}

ztr_un_vote_recalculate_results = {
	ztr_un_vote_clear_results = yes
	set_global_variable = {
		name = ztr_un_vote_support_count
		value = 0
	}
	set_global_variable = {
		name = ztr_un_vote_neutral_count
		value = 0
	}
	set_global_variable = {
		name = ztr_un_vote_against_count
		value = 0
	}
	set_global_variable = {
		name = ztr_un_vote_total_count
		value = 0
	}
	every_scope_je_involved = {
		# Use vote weight if set (Weighted by Prestige model), otherwise default to 1
		if = {
			limit = {
				has_variable = ztr_un_vote_weight
			}
			change_global_variable = {
				name = ztr_un_vote_total_count
				add = var:ztr_un_vote_weight
			}
			if = {
				limit = {
					has_variable = ztr_un_vote_choice_support
				}
				change_global_variable = {
					name = ztr_un_vote_support_count
					add = var:ztr_un_vote_weight
				}
			}
			else_if = {
				limit = {
					has_variable = ztr_un_vote_choice_against
				}
				change_global_variable = {
					name = ztr_un_vote_against_count
					add = var:ztr_un_vote_weight
				}
			}
			else = {
				change_global_variable = {
					name = ztr_un_vote_neutral_count
					add = var:ztr_un_vote_weight
				}
			}
		}
		else = {
			change_global_variable = {
				name = ztr_un_vote_total_count
				add = 1
			}
			if = {
				limit = {
					has_variable = ztr_un_vote_choice_support
				}
				change_global_variable = {
					name = ztr_un_vote_support_count
					add = 1
				}
			}
			else_if = {
				limit = {
					has_variable = ztr_un_vote_choice_against
				}
				change_global_variable = {
					name = ztr_un_vote_against_count
					add = 1
				}
			}
			else = {
				change_global_variable = {
					name = ztr_un_vote_neutral_count
					add = 1
				}
			}
		}
	}
	set_global_variable = {
		name = ztr_un_vote_support_percent
		value = 0
	}
	set_global_variable = {
		name = ztr_un_vote_neutral_percent
		value = 0
	}
	set_global_variable = {
		name = ztr_un_vote_against_percent
		value = 0
	}
	if = {
		limit = {
			global_var:ztr_un_vote_total_count > 0
		}
		set_global_variable = {
			name = ztr_un_vote_support_percent
			value = global_var:ztr_un_vote_support_count
		}
		change_global_variable = {
			name = ztr_un_vote_support_percent
			divide = global_var:ztr_un_vote_total_count
		}
		change_global_variable = {
			name = ztr_un_vote_support_percent
			multiply = 100
		}
		set_global_variable = {
			name = ztr_un_vote_neutral_percent
			value = global_var:ztr_un_vote_neutral_count
		}
		change_global_variable = {
			name = ztr_un_vote_neutral_percent
			divide = global_var:ztr_un_vote_total_count
		}
		change_global_variable = {
			name = ztr_un_vote_neutral_percent
			multiply = 100
		}
		set_global_variable = {
			name = ztr_un_vote_against_percent
			value = global_var:ztr_un_vote_against_count
		}
		change_global_variable = {
			name = ztr_un_vote_against_percent
			divide = global_var:ztr_un_vote_total_count
		}
		change_global_variable = {
			name = ztr_un_vote_against_percent
			multiply = 100
		}
	}
}

# Start a UN vote on a governance proposal
# Expects ztr_un_vote_subject and ztr_un_vote_proposal to be set before calling
ztr_un_start_governance_vote = {
	# One summary line instead of a trigger_event line per member
	custom_tooltip = ztr_un_start_governance_vote_tt
	hidden_effect = {
		set_global_variable = ztr_un_vote_in_progress
		set_global_variable = {
			name = ztr_un_vote_months_remaining
			value = 6
		}
		every_country = {
			limit = {
				has_variable = ztr_un_vote_eligible
			}
			remove_variable = ztr_un_vote_eligible
			remove_variable = ztr_un_vote_choice_support
			remove_variable = ztr_un_vote_choice_neutral
			remove_variable = ztr_un_vote_choice_against
			remove_variable = ztr_un_vote_weight
			remove_variable = ztr_un_vote_proposer
			remove_variable = ztr_un_vote_cosigner
			remove_variable = ztr_un_vote_veto
			remove_variable = ztr_un_vote_committed
			remove_variable = ztr_un_vote_lean
		}
		# Model 1: One Country One Vote - all UN members
		if = {
			limit = {
				global_var:ztr_un_law_voting_model = 1
			}
			every_country = {
				limit = {
					OR = {
						has_modifier = ztr_onu_member
						has_modifier = ztr_onu_permanent_member
					}
					NOT = {
						ztr_un_is_suspended = yes
					}
				}
				set_variable = ztr_un_vote_eligible
				set_variable = {
					name = ztr_un_vote_weight
					value = 1
				}
				trigger_event = {
					id = ztr_un_event.10
					days = 30
				}
			}
		}
		# Model 2: Weighted by Prestige - all UN members, weight by rank
		else_if = {
			limit = {
				global_var:ztr_un_law_voting_model = 2
			}
			every_country = {
				limit = {
					OR = {
						has_modifier = ztr_onu_member
						has_modifier = ztr_onu_permanent_member
					}
					NOT = {
						ztr_un_is_suspended = yes
					}
				}
				set_variable = ztr_un_vote_eligible
				# Great powers get 25 votes, major powers 10, minor powers 5, others 1
				if = {
					limit = {
						country_rank = rank_value:great_power
					}
					set_variable = {
						name = ztr_un_vote_weight
						value = 25
					}
				}
				else_if = {
					limit = {
						country_rank = rank_value:major_power
					}
					set_variable = {
						name = ztr_un_vote_weight
						value = 10
					}
				}
				else_if = {
					limit = {
						country_rank = rank_value:minor_power
					}
					set_variable = {
						name = ztr_un_vote_weight
						value = 5
					}
				}
				else = {
					set_variable = {
						name = ztr_un_vote_weight
						value = 1
					}
				}
				trigger_event = {
					id = ztr_un_event.10
					days = 30
				}
			}
		}
		# Model 3: Security Council Only - permanent and elected members (veto stays with permanent members)
		else_if = {
			limit = {
				global_var:ztr_un_law_voting_model = 3
			}
			every_country = {
				limit = {
					OR = {
						has_modifier = ztr_onu_permanent_member
						ztr_un_is_elected_member = yes
					}
					NOT = {
						ztr_un_is_suspended = yes
					}
				}
				set_variable = ztr_un_vote_eligible
				set_variable = {
					name = ztr_un_vote_weight
					value = 1
				}
				trigger_event = {
					id = ztr_un_event.10
					days = 30
				}
			}
		}
		# Model 4: Superpowers Only - top 2 ranked countries in UN
		else_if = {
			limit = {
				global_var:ztr_un_law_voting_model = 4
			}
			ordered_country = {
				limit = {
					OR = {
						has_modifier = ztr_onu_member
						has_modifier = ztr_onu_permanent_member
					}
				}
				order_by = prestige
				max = 2
				check_range_bounds = no
				# A suspended superpower loses its vote, the seat is not passed on
				if = {
					limit = {
						NOT = {
							ztr_un_is_suspended = yes
						}
					}
					set_variable = ztr_un_vote_eligible
					set_variable = {
						name = ztr_un_vote_weight
						value = 1
					}
					trigger_event = {
						id = ztr_un_event.10
						days = 30
					}
				}
			}
		}
	}
}

# Apply the voted governance change (called from je_un_votations on_timeout when vote passes)
ztr_un_apply_governance_vote = {
	# Subject 1: Voting Model
	if = {
		limit = {
			has_global_variable = ztr_un_vote_subject
			global_var:ztr_un_vote_subject = 1
		}
		remove_global_variable = ztr_un_law_voting_model
		set_global_variable = {
			name = ztr_un_law_voting_model
			value = global_var:ztr_un_vote_proposal
		}
	}
	# Subject 2: Veto Right
	if = {
		limit = {
			has_global_variable = ztr_un_vote_subject
			global_var:ztr_un_vote_subject = 2
		}
		remove_global_variable = ztr_un_law_veto_right
		set_global_variable = {
			name = ztr_un_law_veto_right
			value = global_var:ztr_un_vote_proposal
		}
	}
	# Subject 3: Member Selection - paradigm shift
	if = {
		limit = {
			has_global_variable = ztr_un_vote_subject
			global_var:ztr_un_vote_subject = 3
		}
		remove_global_variable = ztr_un_law_member_selection
		set_global_variable = {
			name = ztr_un_law_member_selection
			value = global_var:ztr_un_vote_proposal
		}
		# Strip all current permanent members to regular members
		every_country = {
			limit = {
				has_modifier = ztr_onu_permanent_member
			}
			remove_modifier = ztr_onu_permanent_member
			add_modifier = ztr_onu_member
		}
		# Select new permanent members based on the new model
		ztr_un_select_new_permanent_members = yes
		# New permanent members: elected seats are drawn again to avoid duplicates
		ztr_un_elect_seats = yes
	}
	# Subject 4: Human Rights - Decolonization
	if = {
		limit = {
			has_global_variable = ztr_un_vote_subject
			global_var:ztr_un_vote_subject = 4
		}
		if = {
			limit = {
				has_global_variable = ztr_un_hr_decolonization
			}
			remove_global_variable = ztr_un_hr_decolonization
		}
		set_global_variable = {
			name = ztr_un_hr_decolonization
			value = global_var:ztr_un_vote_proposal
		}
		#Apply effects based on the stance
		# Decolonization Stance
		# Conservative Decolonization
		if = {
			limit = {
				global_var:ztr_un_hr_decolonization = 1
			}
			every_country = {
				limit = {
					ztr_is_un_member = yes
					OR = {
						has_law_or_variant = law_type:law_colonial_resettlement
						has_law_or_variant = law_type:law_colonial_exploitation
					}
				}
				activate_law = law_type:law_no_colonial_affairs
			}
		}
		# Encouraged Decolonization
		else_if = {
			limit = {
				global_var:ztr_un_hr_decolonization = 2
			}
			#TODO Pensare se dover aggiungere altro
			every_country = {
				limit = {
					ztr_is_un_member = yes
					OR = {
						has_law_or_variant = law_type:law_colonial_resettlement
						has_law_or_variant = law_type:law_colonial_exploitation
						has_law_or_variant = law_type:law_frontier_colonization
					}
				}
				activate_law = law_type:law_no_colonial_affairs
			}
		}
	}
	# Subject 5: Human Rights - Slavery
	if = {
		limit = {
			has_global_variable = ztr_un_vote_subject
			global_var:ztr_un_vote_subject = 5
		}
		if = {
			limit = {
				has_global_variable = ztr_un_hr_slavery
			}
			remove_global_variable = ztr_un_hr_slavery
		}
		set_global_variable = {
			name = ztr_un_hr_slavery
			value = global_var:ztr_un_vote_proposal
		}
		#Apply effects based on the stance
		# Illegal: add modifier to all UN members that disables slavery laws
		if = {
			limit = {
				global_var:ztr_un_hr_slavery = 1
			}
			every_country = {
				limit = {
					ztr_is_un_member = yes
					OR = {
						has_law_or_variant = law_type:law_colonial_slavery
						has_law_or_variant = law_type:law_debt_slavery
						has_law_or_variant = law_type:law_slave_trade
						has_law_or_variant = law_type:law_legacy_slavery
					}
				}
				activate_law = law_type:law_slavery_banned
				set_variable = {
					name = ztr_recently_banned_slavery
					years = 10
				}
			}
		}
	}
	# Subject 6: Human Rights - Children's Rights
	if = {

# Snapshot of a closing UN vote for ztr_un_event.11 (survives ztr_un_vote_cleanup_all_members)
# ztr_un_last_vote_result: 1 adopted, 2 rejected by the majority, 3 majority blocked by a veto
ztr_un_vote_save_last_result = {
	set_global_variable = {
		name = ztr_un_last_vote_subject
		value = global_var:ztr_un_vote_subject
	}
	set_global_variable = {
		name = ztr_un_last_vote_proposal
		value = global_var:ztr_un_vote_proposal
	}
	set_global_variable = {
		name = ztr_un_last_vote_support
		value = global_var:ztr_un_vote_support_count
	}
	set_global_variable = {
		name = ztr_un_last_vote_neutral
		value = global_var:ztr_un_vote_neutral_count
	}
	set_global_variable = {
		name = ztr_un_last_vote_against
		value = global_var:ztr_un_vote_against_count
	}
	set_global_variable = {
		name = ztr_un_last_vote_veto_count
		value = 0
	}
	# Seat candidacy: names for the result event and the UN panel
	remove_global_variable = ztr_un_last_vote_candidate
	remove_global_variable = ztr_un_last_vote_challenged
	if = {
		limit = {
			has_global_variable = ztr_un_vote_candidate
		}
		set_global_variable = {
			name = ztr_un_last_vote_candidate
			value = global_var:ztr_un_vote_candidate
		}
	}
	if = {
		limit = {
			has_global_variable = ztr_un_vote_challenged
		}
		set_global_variable = {
			name = ztr_un_last_vote_challenged
			value = global_var:ztr_un_vote_challenged
		}
	}
	if = {
		limit = {
			global_var:ztr_un_law_veto_right = 1
		}
		# A vote against is not a veto: only a declared veto (ztr_un_vote_veto) blocks
		every_country = {
			limit = {
				has_modifier = ztr_onu_permanent_member
				has_variable = ztr_un_vote_choice_against
				has_variable = ztr_un_vote_veto
				NOT = {
					ztr_un_is_vote_challenged = yes
				}
			}
			change_global_variable = {
				name = ztr_un_last_vote_veto_count
				add = 1
			}
		}
		ordered_country = {
			limit = {
				has_modifier = ztr_onu_permanent_member
				has_variable = ztr_un_vote_choice_against
				has_variable = ztr_un_vote_veto
				NOT = {
					ztr_un_is_vote_challenged = yes
				}
			}
			order_by = prestige
			set_global_variable = {
				name = ztr_un_last_vote_vetoer
				value = this
			}
		}
	}
	if = {
		limit = {
			NOT = {
				global_var:ztr_un_vote_support_count > global_var:ztr_un_vote_against_count
			}
		}
		set_global_variable = {
			name = ztr_un_last_vote_result
			value = 2
		}
	}
	else_if = {
		limit = {
			global_var:ztr_un_last_vote_veto_count > 0
		}
		set_global_variable = {
			name = ztr_un_last_vote_result
			value = 3
		}
		# A veto that blocked a majority recharges in 10 years
		every_country = {
			limit = {
				has_modifier = ztr_onu_permanent_member
				has_variable = ztr_un_vote_choice_against
				has_variable = ztr_un_vote_veto
				NOT = {
					ztr_un_is_vote_challenged = yes
				}
			}
			set_variable = {
				name = ztr_un_veto_cd
				years = 10
			}
		}
	}
	else = {
		set_global_variable = {
			name = ztr_un_last_vote_result
			value = 1
		}
	}
}

# Yearly AI proposal coordinator (global scope): seat candidacy, then permanent members, then elected members
ztr_un_ai_yearly_proposal = {
	if = {
		limit = {
			NOR = {
				has_global_variable = ztr_un_vote_in_progress
				has_global_variable = ztr_un_vote_cd
				# A player is choosing a proposal in the menu events or deciding on a co-signature
				has_global_variable = ztr_un_button_opened
			}
		}
		# AI candidacy for a permanent seat, half of the years it is possible
		random_country = {
			limit = {
				is_ai = yes
				ztr_un_can_run_for_permanent_seat = yes
			}
			save_scope_as = ztr_un_ai_seat_candidate
		}
		if = {
			limit = {
				exists = scope:ztr_un_ai_seat_candidate
			}
			random = {
				chance = 50
				scope:ztr_un_ai_seat_candidate = {
					ztr_un_start_seat_candidacy = yes
				}
			}
		}
		# Up to two AI permanent members try one proposal each
		if = {
			limit = {
				NOT = {
					has_global_variable = ztr_un_vote_in_progress
				}
			}
			random_country = {
				limit = {
					is_ai = yes
					has_modifier = ztr_onu_permanent_member
					NOT = {
						ztr_un_is_suspended = yes
					}
				}
				save_scope_as = ztr_un_potential_proposer_scope_1
			}
			if = {
				limit = {
					exists = scope:ztr_un_potential_proposer_scope_1
				}
				random_country = {
					limit = {
						is_ai = yes
						has_modifier = ztr_onu_permanent_member
						NOT = {
							this = scope:ztr_un_potential_proposer_scope_1
						}
						NOT = {
							ztr_un_is_suspended = yes
						}
					}
					save_scope_as = ztr_un_potential_proposer_scope_2
				}
				scope:ztr_un_potential_proposer_scope_1 = {
					ztr_un_random_subject_and_law_extraction = yes
					ztr_un_ai_try_start_possible_vote = yes
				}
				# Second proposer evaluates the same extracted proposal
				if = {
					limit = {
						NOT = {
							has_global_variable = ztr_un_vote_in_progress
						}
						exists = scope:ztr_un_potential_proposer_scope_2
					}
					scope:ztr_un_potential_proposer_scope_2 = {
						ztr_un_ai_try_start_possible_vote = yes
					}
				}
			}
		}
		# One AI elected member tries a human rights statement, which needs a co-signer
		if = {
			limit = {
				NOT = {
					has_global_variable = ztr_un_vote_in_progress
				}
			}
			random_country = {
				limit = {
					is_ai = yes
					ztr_un_elected_can_propose = yes
				}
				ztr_un_random_human_rights_extraction = yes
				ztr_un_ai_elected_try_proposal = yes
			}
		}
	}
}

# 4 random seats for 9 years; holders of the previous round cannot be drawn again
ztr_un_elect_seats = {
	every_country = {
		limit = {
			has_variable = ztr_un_elected_previous
		}
		remove_variable = ztr_un_elected_previous
	}
	every_country = {
		limit = {
			OR = {
				has_variable = ztr_un_elected_seat
				has_variable = ztr_un_elected_seat_lost
			}
		}
		remove_variable = ztr_un_elected_seat
		remove_variable = ztr_un_elected_seat_lost
		set_variable = ztr_un_elected_previous
	}
	random_country = {
		limit = {
			ztr_un_can_be_elected = yes
		}
		ztr_un_take_elected_seat = yes
	}
	random_country = {
		limit = {
			ztr_un_can_be_elected = yes
		}
		ztr_un_take_elected_seat = yes
	}
	random_country = {
		limit = {
			ztr_un_can_be_elected = yes
		}
		ztr_un_take_elected_seat = yes
	}
	random_country = {
		limit = {
			ztr_un_can_be_elected = yes
		}
		ztr_un_take_elected_seat = yes
	}
	set_global_variable = {
		name = ztr_un_elected_term
		years = 9
	}
}

ztr_un_take_elected_seat = {
	set_variable = ztr_un_elected_seat
	if = {
		limit = {
			is_ai = no
		}
		trigger_event = {
			id = ztr_un_event.39
			popup = yes
		}
	}
}

# Seat lost mid-term (condemnation, expulsion, withdrawal): still counts as the previous round
# By-election: a random country from the same pool takes the seat for the rest of the term
ztr_un_remove_elected_seat = {
	if = {
		limit = {
			has_variable = ztr_un_elected_seat
		}
		remove_variable = ztr_un_elected_seat
		set_variable = ztr_un_elected_seat_lost
		random_country = {
			limit = {
				ztr_un_can_be_elected = yes
				NOT = {
					has_variable = ztr_un_elected_seat_lost
				}
			}
			ztr_un_take_elected_seat = yes
		}
	}
}

# PERMANENT SEAT CANDIDACY (subject 15)
# ========================================
# Scope: the candidate. The challenged member is the weakest permanent under the current law; none if a seat is vacant
ztr_un_start_seat_candidacy = {
	set_global_variable = {
		name = ztr_un_vote_subject
		value = 15
	}
	set_global_variable = {
		name = ztr_un_vote_proposal
		value = 1
	}
	set_global_variable = {
		name = ztr_un_vote_candidate
		value = this
	}
	remove_global_variable = ztr_un_vote_challenged
	# By Prestige: lowest prestige among 5 permanent members
	if = {
		limit = {
			global_var:ztr_un_law_member_selection = 1
			any_country = {
				count >= 5
				has_modifier = ztr_onu_permanent_member
			}
		}
		ordered_country = {
			limit = {
				has_modifier = ztr_onu_permanent_member
			}
			order_by = {
				value = 0
				subtract = prestige
			}
			set_global_variable = {
				name = ztr_un_vote_challenged
				value = this
			}
		}
	}
	# By Military Power: lowest army + navy power projection among 5 permanent members
	else_if = {
		limit = {
			global_var:ztr_un_law_member_selection = 3
			any_country = {
				count >= 5
				has_modifier = ztr_onu_permanent_member
			}
		}
		ordered_country = {
			limit = {
				has_modifier = ztr_onu_permanent_member
			}
			order_by = {
				value = 0
				subtract = army_power_projection
				subtract = naval_power_projection
			}
			set_global_variable = {
				name = ztr_un_vote_challenged
				value = this
			}
		}
	}
	# Superpowers: the weakest permanent member outside the top 2
	else_if = {
		limit = {
			global_var:ztr_un_law_member_selection = 4
			any_country = {
				count >= 2
				has_modifier = ztr_onu_permanent_member
			}
		}
		ordered_country = {
			limit = {
				has_modifier = ztr_onu_permanent_member
				global_country_ranking > 2
			}
			order_by = {
				value = 0
				subtract = prestige
			}
			set_global_variable = {
				name = ztr_un_vote_challenged
				value = this
			}
		}
	}
	set_variable = {
		name = ztr_un_seat_candidacy_cd
		years = 30
	}
	ztr_un_start_governance_vote = yes
	set_variable = ztr_un_vote_proposer
}

# ELECTED MEMBER PROPOSALS (co-signed by a permanent member)
# ========================================
# Human rights menu options: elected members look for a co-signer, permanent members start the vote
ztr_un_submit_human_rights_proposal = {
	if = {
		limit = {
			has_variable = ztr_un_elected_proposing
		}
		custom_tooltip = ztr_un_submit_cosign_tt
		hidden_effect = {
			remove_variable = ztr_un_elected_proposing
			ztr_un_request_cosign = yes
		}
	}
	else = {
		ztr_un_start_governance_vote = yes
		set_variable = ztr_un_vote_proposer
		remove_global_variable = ztr_un_button_opened
	}
}

# Scope: elected proposer, with ztr_un_vote_subject and ztr_un_vote_proposal already set
# AI permanent members that back the proposal or the proposer co-sign at once; otherwise a player permanent member is asked
ztr_un_request_cosign = {
	save_scope_as = ztr_un_cosign_proposer
	every_country = {
		limit = {
			is_ai = yes
			has_modifier = ztr_onu_permanent_member
			NOT = {
				ztr_un_is_suspended = yes
			}
		}
		ztr_un_ai_evaluate_cosign = yes
	}
	if = {
		limit = {
			any_country = {
				has_variable = ztr_un_cosign_willing
			}
		}
		ordered_country = {
			limit = {
				has_variable = ztr_un_cosign_willing
			}
			order_by = var:ztr_un_cosign_score
			save_scope_as = ztr_un_cosigner
		}
		ztr_un_start_cosigned_vote = yes
	}
	else_if = {
		limit = {
			any_country = {
				is_ai = no
				has_modifier = ztr_onu_permanent_member
				NOT = {
					ztr_un_is_suspended = yes
				}
			}
		}
		set_global_variable = ztr_un_button_opened
		set_global_variable = {
			name = ztr_un_cosign_pending
			value = this
		}
		random_country = {
			limit = {
				is_ai = no
				has_modifier = ztr_onu_permanent_member
				NOT = {
					ztr_un_is_suspended = yes
				}
			}
			trigger_event = {
				id = ztr_un_event.37
				popup = yes
			}
		}
	}
	else = {
		ztr_un_elected_proposal_failed = yes
	}
	every_country = {
		limit = {
			has_variable = ztr_un_cosign_willing
		}
		remove_variable = ztr_un_cosign_willing
	}
	every_country = {
		limit = {
			has_variable = ztr_un_cosign_score
		}
		remove_variable = ztr_un_cosign_score
	}
}

# Needs scope:ztr_un_cosign_proposer and scope:ztr_un_cosigner
ztr_un_start_cosigned_vote = {
	ztr_un_start_governance_vote = yes
	set_global_variable = {
		name = ztr_un_last_cosigner
		value = scope:ztr_un_cosigner
	}
	set_global_variable = {
		name = ztr_un_last_cosign_result
		value = 1
	}
	scope:ztr_un_cosigner = {
		set_variable = ztr_un_vote_proposer
		# Not the author: its political pole does not sway the vote
		set_variable = ztr_un_vote_cosigner
	}
	scope:ztr_un_cosign_proposer = {
		set_variable = ztr_un_vote_proposer
		set_variable = {
			name = ztr_un_elected_proposal_cd
			years = 10
		}
		if = {
			limit = {
				is_ai = no
			}
			trigger_event = {
				id = ztr_un_event.38
				popup = yes
			}
		}
	}
	set_global_variable = {
		name = ztr_un_elected_proposal_gap
		years = 1
	}
	remove_global_variable = ztr_un_cosign_pending
	remove_global_variable = ztr_un_button_opened
}

# No co-signer: the proposer may try again in a year
ztr_un_elected_proposal_failed = {
	set_global_variable = {
		name = ztr_un_last_cosign_result
		value = 0
	}
	scope:ztr_un_cosign_proposer = {
		set_variable = {
			name = ztr_un_elected_proposal_cd
			years = 1
		}
		if = {
			limit = {
				is_ai = no
			}
			trigger_event = {
				id = ztr_un_event.38
				popup = yes
			}
		}
	}
	remove_global_variable = ztr_un_vote_subject
	remove_global_variable = ztr_un_vote_proposal
	remove_global_variable = ztr_un_cosign_pending
	remove_global_variable = ztr_un_button_opened
}

# UN BUDGET (dues)
# ========================================
# Start of the year: weekly dues per member (ztr_un_dues_annual / 52) and the yearly UN budget
ztr_un_assess_dues = {
	set_global_variable = {
		name = ztr_un_budget
		value = 0
	}
	every_country = {
		limit = {
			has_variable = ztr_un_dues_weekly
		}
		remove_variable = ztr_un_dues_weekly
	}
	every_country = {
		limit = {
			ztr_is_un_member = yes
		}
		set_variable = {
			name = ztr_un_dues_weekly
			value = ztr_un_dues_annual
		}
		change_variable = {
			name = ztr_un_dues_weekly
			divide = 52
		}
		if = {
			limit = {
				NOT = {
					ztr_un_is_suspended = yes
				}
			}
			change_global_variable = {
				name = ztr_un_budget
				add = ztr_un_dues_annual
			}
		}
	}
}

# Scope: the committed voter. $CHOICE$ = support or against
ztr_un_commit_vote = {
	remove_variable = ztr_un_vote_choice_neutral
	remove_variable = ztr_un_vote_choice_support
	remove_variable = ztr_un_vote_choice_against
	remove_variable = ztr_un_vote_veto
	set_variable = ztr_un_vote_choice_$CHOICE$
	set_variable = ztr_un_vote_committed
	je:je_un_votations ?= {
		ztr_un_vote_update_results = yes
	}
}
