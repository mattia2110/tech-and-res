				has_modifier = nuclear_damage_1
			}
		}
		if = { 
			limit = { 
				has_modifier = nuclear_damage_5
				OR = {
					has_modifier = nuclear_damage_4

					has_modifier = nuclear_damage_2
					has_modifier = nuclear_damage_1
				}
			}

			remove_modifier = nuclear_damage_3
			remove_modifier = nuclear_damage_2
			remove_modifier = nuclear_damage_1
		}
		if = { 
			limit = { 
				has_modifier = nuclear_damage_4
				OR = {
					has_modifier = nuclear_damage_3
					has_modifier = nuclear_damage_2
					has_modifier = nuclear_damage_1
				}
			}
			remove_modifier = nuclear_damage_3

# One strike; ATTACKER and DEFENDER are explicit country scopes.
# Cooldown is checked before an exchange, allowing its salvo to reach several targets.
ztr_apply_nuclear_strike = {
	if = {
		limit = {
			$ATTACKER$ = {
				has_modifier = nuclear_capability
				ztr_has_operational_nuclear_silo = yes
				ztr_nuclear_target_in_range = { TARGET = $DEFENDER$ }
				NOT = { this = $DEFENDER$ }
				ztr_nuclear_attack_strength > 0
			}
		}
		$ATTACKER$ = { save_scope_as = nuclear_war_actor }
		$DEFENDER$ = {
			save_scope_as = nuclear_war_target
			set_variable = { name = ztr_nuclear_target_defence value = ztr_nuclear_defence_strength }
		}
		every_country = {
			limit = {
				OR = {
					has_diplomatic_pact = { who = $ATTACKER$ type = station_nuclear_weapons is_initiator = no }
					is_subject_of = $ATTACKER$
				}
				has_strategic_adjacency = $DEFENDER$
				NOT = { this = $DEFENDER$ }
			}
			$DEFENDER$ = { change_variable = { name = ztr_nuclear_target_defence multiply = 0.9 } }
		}
		$DEFENDER$ = {
			set_variable = {
				name = ztr_nuclear_damage_ratio
				value = {
					value = $ATTACKER$.ztr_nuclear_attack_strength
					divide = { value = var:ztr_nuclear_target_defence min = 0.01 }
				}
			}
			custom_tooltip = {
				text = launch_nuclear_weapons.1.tt
				if = {
					limit = { var:ztr_nuclear_damage_ratio <= 1 }
					effect_tiny_nuclear_damage = yes
				}
				else_if = {
					limit = { var:ztr_nuclear_damage_ratio <= 2 }
					effect_very_low_nuclear_damage = yes
				}
				else_if = {
					limit = { var:ztr_nuclear_damage_ratio <= 4 }
					effect_low_nuclear_damage = yes
				}
				else_if = {
					limit = { var:ztr_nuclear_damage_ratio <= 6 }
					effect_medium_nuclear_damage = yes
				}
				else = { effect_high_nuclear_damage = yes }
			}
			effect_cleanup_nuclear_damage = yes
			remove_variable = ztr_nuclear_damage_ratio
			remove_variable = ztr_nuclear_target_defence
		}
		$ATTACKER$ = {
			remove_modifier = nuclear_attack_cooldown
			add_modifier = { name = nuclear_attack_cooldown years = 1 }
		}
		# Capture the first impact before any retaliation overwrites presentation scopes.
		if = {
			limit = { NOT = { has_global_variable = ztr_atomic_bomb_first_use_var } }
			set_global_variable = ztr_atomic_bomb_first_use_var
			$ATTACKER$ = { save_scope_as = ztr_first_nuclear_attacker }
			$DEFENDER$ = { save_scope_as = ztr_first_nuclear_defender }
			scope:nuclear_striked_state_target = { save_scope_as = ztr_first_nuclear_state }
			every_country = {
				limit = {
					NOR = { this = $ATTACKER$ this = $DEFENDER$ }
				}
				trigger_event = nuclear_warfare_system.9
			}
		}
	}
}

ztr_nuclear_strike_and_response = {
	if = {
		limit = {
			$ATTACKER$ = {
				ztr_nuclear_attack_strength > 0
				ztr_nuclear_target_in_range = { TARGET = $DEFENDER$ }
				NOT = { this = $DEFENDER$ }
			}
		}
		ztr_apply_nuclear_strike = { ATTACKER = $ATTACKER$ DEFENDER = $DEFENDER$ }
		if = {
			limit = { $DEFENDER$ = { ztr_can_retaliate_nuclear_strike = { TARGET = $ATTACKER$ } } }
			ztr_apply_nuclear_strike = { ATTACKER = $DEFENDER$ DEFENDER = $ATTACKER$ }
		}
	}
}

# Diplomatic action entry point. The saved leaders remain fixed for the whole exchange.
ztr_launch_nuclear_exchange = {
	$ATTACKER$ = { save_scope_as = ztr_nuclear_exchange_attacker }
	$DEFENDER$ = { save_scope_as = ztr_nuclear_exchange_defender }
	if = {
		limit = { $DEFENDER$ = { ztr_nuclear_arsenal_ready = yes } }
		$DEFENDER$ = { save_scope_as = ztr_nuclear_retaliating_leader }
	}
	# The protected war leader may be hosting an arsenal owned by a nonparticipant.
	if = {
		limit = { $ATTACKER$ = { has_war_with = $DEFENDER$ } }
		$ATTACKER$ = { save_scope_as = ztr_nuclear_protected_warleader }
	}
	else = {
		random_country = {
			limit = {
				has_diplomatic_pact = { who = $ATTACKER$ type = station_nuclear_weapons is_initiator = no }
				has_war_with = $DEFENDER$
				any_scope_war = { is_warleader = prev is_warleader = $DEFENDER$ }
			}
			save_scope_as = ztr_nuclear_protected_warleader
		}
	}
	ztr_nuclear_strike_and_response = {
		ATTACKER = scope:ztr_nuclear_exchange_attacker
		DEFENDER = scope:ztr_nuclear_exchange_defender
	}
	if = {
		limit = { always = $ESCALATE$ exists = scope:ztr_nuclear_protected_warleader }
		every_country = {
			limit = {
				has_war_with = scope:ztr_nuclear_protected_warleader
				NOT = { this = scope:ztr_nuclear_exchange_defender }
				is_revolutionary = no
				is_secessionist = no
				any_scope_war = {
					is_war_participant = scope:ztr_nuclear_protected_warleader
					is_war_participant = scope:ztr_nuclear_exchange_defender
				}
			}
			save_scope_as = ztr_nuclear_secondary_target
			if = {
				limit = {
					scope:ztr_nuclear_exchange_attacker = {
						ztr_nuclear_target_in_range = { TARGET = scope:ztr_nuclear_secondary_target }
					}
				}
				ztr_nuclear_strike_and_response = {
					ATTACKER = scope:ztr_nuclear_exchange_attacker
					DEFENDER = scope:ztr_nuclear_secondary_target
				}
			}
		}
		if = {
			limit = { exists = scope:ztr_nuclear_retaliating_leader }
			every_country = {
				limit = {
					has_war_with = scope:ztr_nuclear_exchange_defender
					NOT = { this = scope:ztr_nuclear_exchange_attacker }
					is_revolutionary = no
					is_secessionist = no
					any_scope_war = {
						is_war_participant = scope:ztr_nuclear_protected_warleader
						is_war_participant = scope:ztr_nuclear_exchange_defender
					}
				}
				save_scope_as = ztr_nuclear_secondary_target
				if = {
					limit = {
						scope:ztr_nuclear_exchange_defender = {
							ztr_nuclear_target_in_range = { TARGET = scope:ztr_nuclear_secondary_target }
						}
					}
					ztr_nuclear_strike_and_response = {
						ATTACKER = scope:ztr_nuclear_exchange_defender
						DEFENDER = scope:ztr_nuclear_secondary_target
					}
				}
			}
		}
	}
	ztr_refresh_nuclear_scores = yes
}

calculate_nuclear_strength_info = {
	set_variable = { name = nuclear_strike_power_display value = ztr_nuclear_attack_strength }
}

calculate_nuclear_defence_info = {
	set_variable = { name = nuclear_strike_defence_display value = ztr_nuclear_defence_strength }
}

# One world scan per annual pulse or nuclear exchange, not one per country panel.
ztr_refresh_nuclear_scores = {
	set_global_variable = ztr_nuclear_scores_initialized
	every_country = {
		limit = {
			OR = { has_modifier = nuclear_capability has_variable = nuclear_strike_power_display }
		}
		calculate_nuclear_strength_info = yes
		calculate_nuclear_defence_info = yes
	}
	every_country = {
		limit = { has_journal_entry = je_nuclear_weapons_info_panel }
		ztr_nuclear_ranking_variables = yes
		ztr_nuclear_strength_ranking_effect = yes
		ztr_nuclear_defence_ranking_effect = yes
	}
}
