# Here we have all the functions needed for UN
#1. ztr_un_random_subject_and_law_extraction -> Randomly selects a subject and a law for the UN to vote on
#2. ztr_un_set_possible_law_as_considered -> Sets the possible law as considered
#3. ztr_un_set_current_law_as_considered -> Sets the current law as considered
#4. ztr_un_set_vote_proposal_as_considered -> Sets the active vote proposal as considered
#5. ztr_un_set_current_vote_law_as_considered -> Sets the current law of the active vote subject as considered
#6. ztr_un_prioritize_first_institution_establishment_vote -> Once per institution, prioritizes an available UN institution establishment proposal
#7. ztr_un_ai_try_start_possible_vote -> Tries to start a possible vote
#8. ztr_un_ai_take_neutral_vote_position -> Lets a neutral AI country pick support or against during a live vote
#9. ztr_un_value_law_prepare_country_context -> Prepares the country context for the law
#10. ztr_un_value_law -> Used by country to decide favor, neutral or against and by proposer if it wants to propose the law
#11. ztr_un_random_human_rights_extraction -> Like #1, limited to the human rights statements (elected members)
#12. ztr_un_ai_elected_try_proposal -> An AI elected member proposes the extracted statement if it prefers it, then looks for a co-signer
#13. ztr_un_ai_evaluate_cosign -> An AI permanent member decides whether to co-sign an elected member proposal

					global_var:ztr_un_institution_imf = 0

					NOT = {

ztr_un_ai_try_start_possible_vote = {
	# Fallback that after UN creation 100% doesn't do anything
	ztr_un_init_governance_laws = yes

	remove_variable = ztr_un_ai_current_law_value
	remove_variable = ztr_un_ai_possible_law_value
	remove_variable = ztr_un_subject_considered
	remove_variable = ztr_un_law_considered

	ztr_un_prioritize_first_institution_establishment_vote = yes

	if = {
		limit = {
			ztr_un_country_can_propose_possible_law = yes
			ztr_un_is_not_applied_law = yes
		}

		# Current Law Check
		ztr_un_set_current_law_as_considered = yes
		ztr_un_value_law = yes
		set_variable = {
			name = ztr_un_ai_current_law_value
			value = var:ztr_un_law_value_total_considered
		}

		# Possible Law Check
		ztr_un_set_possible_law_as_considered = yes
		ztr_un_value_law = yes

				value = global_var:ztr_un_subject_possible_proposal
			}
			set_global_variable = {
				name = ztr_un_vote_proposal
				value = global_var:ztr_un_law_possible_proposal
			}
			ztr_un_request_cosign = yes
		}
	}

	remove_variable = ztr_un_ai_current_law_value
	remove_variable = ztr_un_ai_possible_law_value
	remove_variable = ztr_un_subject_considered
	remove_variable = ztr_un_law_considered
}

# Scope: AI permanent member; scope:ztr_un_cosign_proposer set. Willing when interest in the statement plus ties to the proposer exceed 10
ztr_un_ai_evaluate_cosign = {
	remove_variable = ztr_un_cosign_score
	remove_variable = ztr_un_cosign_willing
	ztr_un_set_current_vote_law_as_considered = yes
	ztr_un_value_law = yes
	set_variable = {
		name = ztr_un_cosign_score
		value = 0
	}
	change_variable = {
		name = ztr_un_cosign_score
		subtract = var:ztr_un_law_value_total_considered
	}
	ztr_un_set_vote_proposal_as_considered = yes
	ztr_un_value_law = yes
	change_variable = {
		name = ztr_un_cosign_score
		add = var:ztr_un_law_value_total_considered
	}
	if = {
		limit = {
			scope:ztr_un_cosign_proposer = {
				relations:prev >= relations_threshold:friendly
			}
		}
		change_variable = {
			name = ztr_un_cosign_score
			add = 20
		}
	}
	if = {
		limit = {
			scope:ztr_un_cosign_proposer = {
				owes_obligation_to = prev
			}
		}
		change_variable = {
			name = ztr_un_cosign_score
			add = 15
		}
	}
	if = {
		limit = {
			owes_obligation_to = scope:ztr_un_cosign_proposer
		}
		change_variable = {
			name = ztr_un_cosign_score
			add = 15
		}
	}
	if = {
		limit = {
			has_diplomatic_pact = {
				who = scope:ztr_un_cosign_proposer
				type = rivalry
			}
		}
		change_variable = {
			name = ztr_un_cosign_score
			add = -30
		}
	}
	# Political pole of the proposer: +10 same, -10 opposite
	if = {
		limit = {
			ztrpol_same_pole_as = {
				COUNTRY = scope:ztr_un_cosign_proposer
			}
		}
		change_variable = {
			name = ztr_un_cosign_score
			add = 10
		}
	}
	else_if = {
		limit = {
			ztrpol_opposite_pole_to = {
				COUNTRY = scope:ztr_un_cosign_proposer
			}
		}
		change_variable = {
			name = ztr_un_cosign_score
			add = -10
		}
	}
	if = {
		limit = {
			var:ztr_un_cosign_score > 10
		}
		set_variable = ztr_un_cosign_willing
	}
	remove_variable = ztr_un_subject_considered
	remove_variable = ztr_un_law_considered
}
