ztr_activate_communist_single_party_law = {
	activate_law = law_type:law_single_party_state
}

# Hook: single-party law of the Menshevik soviet republic
ztr_activate_menshevik_single_party_law = {
	activate_law = law_type:law_single_party_state
}

#############################
# RUSSIAN REVOLUTION IG ROLES
#############################

# Soviet factions: Trade Unions and Rural Folk in vanilla, BPM socialist IGs with BPM
ztr_russia_save_soviet_ig_scopes = {
	if = {
		limit = {
			BPM_is_active_trigger = yes
		}
		ig:ig_revolutionist_socialists ?= {
			save_scope_as = russia_bolsheviks_scope
		}
		ig:ig_reformist_socialists ?= {
			save_scope_as = russia_mensheviks_scope
		}
		ig:ig_agrarian_populists ?= {
			save_scope_as = russia_left_sr_scope
		}
		# Faction chosen at the First Soviet Congress
		if = {
			limit = {
				has_variable = ztr_menshevik_ideology_var
			}
			scope:russia_mensheviks_scope ?= {
				save_scope_as = russia_soviet_winner_scope
			}
		}
		else_if = {
			limit = {
				has_variable = ztr_agro_socialist_ideology_var
			}
			scope:russia_left_sr_scope ?= {
				save_scope_as = russia_soviet_winner_scope
			}
		}
		else = {
			scope:russia_bolsheviks_scope ?= {
				save_scope_as = russia_soviet_winner_scope
			}
		}
	}
	else = {
		ig:ig_trade_unions ?= {
			save_scope_as = russia_bolsheviks_scope
			save_scope_as = russia_mensheviks_scope
			save_scope_as = russia_soviet_winner_scope
		}
		ig:ig_rural_folk ?= {
			save_scope_as = russia_left_sr_scope
		}
	}
}

# Winning soviet faction joins the government and provides the ruler
ztr_russia_soviet_winner_takes_power = {
	ztr_russia_save_soviet_ig_scopes = yes
	scope:russia_soviet_winner_scope ?= {
		if = {
			limit = {
				is_in_government = no
			}
			add_ruling_interest_group = yes
		}
		leader = {
			set_character_as_ruler = yes
		}
	}
}

# Replaces a generic IG modifier with its current share (scope: interest group)
ztr_bpm_sync_ig_modifier = {
	remove_modifier = $MODIFIER$
	if = {
		limit = {
			$VALUE$ != 0
		}
		add_modifier = {
			name = $MODIFIER$
			multiplier = $VALUE$
		}
	}
}

# Mirrors the IG effects of Tech&Res modifiers onto BPM IGs (scope: country)
