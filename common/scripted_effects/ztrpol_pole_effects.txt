# Tech&Res politics: yearly update of the Liberty/Equality pole, after the index (scope: country)
# Thresholds: liberty 50, equality 70; once set, an axis flips only 1 point past its threshold
# Poles start with mass politics: before political_agitation the country is pre-modern
ztrpol_update_pole = {
	if = {
		limit = {
			has_technology_researched = political_agitation
			has_variable = liberty_index_var
			has_variable = equality_index_var
		}
		if = {
			limit = {
				NOT = {
					has_variable = ztrpol_pole_set
				}
			}
			set_variable = ztrpol_pole_set
			if = {
				limit = {
					var:liberty_index_var >= 50
				}
				set_variable = ztrpol_high_liberty
			}
			if = {
				limit = {
					var:equality_index_var >= 70
				}
				set_variable = ztrpol_high_equality
			}
		}
		else = {
			# Liberty axis
			if = {
				limit = {
					has_variable = ztrpol_high_liberty
					var:liberty_index_var < 49
				}
				remove_variable = ztrpol_high_liberty
			}
			else_if = {
				limit = {
					NOT = {
						has_variable = ztrpol_high_liberty
					}
					var:liberty_index_var >= 51
				}
				set_variable = ztrpol_high_liberty
			}
			# Equality axis
			if = {
				limit = {
					has_variable = ztrpol_high_equality
					var:equality_index_var < 69
				}
				remove_variable = ztrpol_high_equality
			}
			else_if = {
				limit = {
					NOT = {
						has_variable = ztrpol_high_equality
					}
					var:equality_index_var >= 71
				}
				set_variable = ztrpol_high_equality
			}
		}
		# Pole age; on a change, IG pole ideologies stay until ztrpol_ig_ideology_yearly converts them, and a pending change is cancelled
		if = {
			limit = {
				ztrpol_pole_changed = yes
			}
			ztrpol_set_pole_code = yes
			set_variable = {
				name = ztrpol_pole_years
				value = 0
			}
			every_interest_group = {
				remove_variable = ztrpol_tr_id
				remove_variable = ztrpol_tr_chosen
			}
		}
		else = {
			change_variable = {
				name = ztrpol_pole_years
				add = 1
			}
		}
	}
}

# Pole code: 1 Social, 2 Liberal, 3 Collectivist, 4 Reactionary
ztrpol_set_pole_code = {
	if = {
		limit = {
			ztrpol_is_social_pole = yes
		}
		set_variable = {
			name = ztrpol_pole_code
			value = 1
		}
	}
	else_if = {
		limit = {
			ztrpol_is_liberal_pole = yes
		}
		set_variable = {
			name = ztrpol_pole_code
			value = 2
		}
	}
	else_if = {
		limit = {
			ztrpol_is_collectivist_pole = yes
		}
		set_variable = {
			name = ztrpol_pole_code
			value = 3
		}
	}
	else = {
		set_variable = {
			name = ztrpol_pole_code
			value = 4
		}
	}
}

# Great Powers with an Ideological Pole, by prestige, listed in je_liberty_index's description (global)
ztrpol_update_gp_poles = {
	clear_global_variable_list = ztrpol_gp_poles_list
	ordered_country = {
		limit = {
			country_rank >= rank_value:great_power
			has_variable = ztrpol_pole_set
		}
		order_by = prestige
		max = 20
		check_range_bounds = no
		add_to_global_variable_list = {
			name = ztrpol_gp_poles_list
			target = this
		}
	}
}
