	interest_group_leader_trigger = {
		ztrpol_ideology_democratic_socialist_valid_trigger = yes

	non_interest_group_leader_trigger = {
		# Agitator ideologies should disapprove of a current law
		NAND = {
			has_role_of_type = agitator
			owner = {
				has_law_or_variant = law_type:law_cooperative_ownership
				has_law_or_variant = law_type:law_graduated_taxation
				has_law_or_variant = law_type:law_old_age_pension

		interest_group ?= {
			ztrpol_ideology_democratic_socialist_valid_trigger = yes
		}

	interest_group_leader_weight = {
		value = {
			add = 100
			desc = "base_value"
		}
		if = {
			limit = {
				ztrpol_ideology_democratic_socialist_natural_trigger = yes

				value = 50
				desc = "ZTRPOL_NATURAL_IG"
