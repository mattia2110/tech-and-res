
# Demographic Stages
# Each bar fills to 100, so overall progress reads as 0-100 stage I, 100-200 stage II, and so on.
ztr_demographic_0_to_1_progress_bar = {
	name = "ztr_demographic_0_to_1_progress_bar"
	desc = "ztr_demographic_0_to_1_progress_bar_desc"
	default_green = yes
	yearly_progress = {
		if = {
			limit = {
				OR = {
					ztr_demographic_above_dev_0 = yes
					year > 1900
				}
			}
			add = {
				desc = "ztr_demographic_0_to_1_development_tt"
				value = 0.9
			}
		}
		if = {
			limit = {
				average_sol > 10
			}
			add = {
				desc = "ztr_demographic_0_to_1_sol_tt"
				value = 1.2
			}
		}
		if = {
			limit = {
				literacy_rate > 0.3
			}
			add = {
				desc = "ztr_demographic_0_to_1_literacy_tt"
				value = 1.2
			}
		}
		if = {
			limit = {
				ztr_demographic_has_healthcare = yes
			}
			add = {
				desc = "ztr_demographic_healthcare_tt"
				value = 0.6
			}
		}
		if = {
			limit = {
				ztr_demographic_traditional_society = yes
			}
			subtract = {
				desc = "ztr_demographic_traditional_values_tt"
				value = 0.3
			}
		}
	}
	start_value = 0
	min_value = 0
	max_value = 100
}
