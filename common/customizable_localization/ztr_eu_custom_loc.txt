# Name of the organization, by ideological orientation
ztr_eu_org_name = {
	type = country
	text = {
		trigger = {
			has_global_variable = ztr_eu_ideology
			global_var:ztr_eu_ideology = 2
		}
		localization_key = ztr_eu_org_name_communist_loc
	}
	text = {
		trigger = {
			has_global_variable = ztr_eu_ideology
			global_var:ztr_eu_ideology = 3
		}
		localization_key = ztr_eu_org_name_authoritarian_loc
	}
	text = {
		localization_key = ztr_eu_org_name_liberal_loc
	}
}
