# Promotion and demotion lines, hidden at the top and bottom level
ztr_development_promotion_line = {
	type = country
	text = {
		trigger = {
			has_variable = ztr_development_level_var
			var:ztr_development_level_var < 8
		}
		localization_key = ztr_development_promotion_line
	}
	text = {
		trigger = { always = yes }
		localization_key = ztr_development_line_none
	}
}

ztr_development_demotion_line = {
	type = country
	text = {
		trigger = {
			has_variable = ztr_development_level_var
			var:ztr_development_level_var > 0
		}
		localization_key = ztr_development_demotion_line
	}
	text = {
		trigger = { always = yes }
		localization_key = ztr_development_line_none
	}
}
