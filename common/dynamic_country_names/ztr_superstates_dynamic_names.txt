# =====================================================================================
# SUPERSTATE NAMES
# Every superstate is born with the name of its founding coalition. The ideological
# orientation at the moment of its birth is stored on the superstate itself in the
# ztr_superstate_ideology country variable:
#     (unset / 1) = the base liberal name, taken straight from the country definition
#     2           = the Commune, a socialist union
#     3           = the Imperium, an authoritarian union
# The same variable drives the flag variants, see ztr_superstates_flags.txt.
# =====================================================================================

# European Federation
EUF = {
	dynamic_country_name = {
		name = dyn_c_european_commune
		adjective = dyn_c_european_commune_adj

		priority = 100

		trigger = {
			has_variable = ztr_superstate_ideology
			var:ztr_superstate_ideology = 2
		}
	}
	dynamic_country_name = {
		name = dyn_c_european_imperium
		adjective = dyn_c_european_imperium_adj

		priority = 100

		trigger = {
			has_variable = ztr_superstate_ideology
			var:ztr_superstate_ideology = 3
		}
	}
}
