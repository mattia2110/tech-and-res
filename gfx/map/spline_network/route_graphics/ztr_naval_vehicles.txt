# Cargo ships on the naval routes (models: scripts/city_gen/ztr_vehiclegen.py). The vanilla sail, sail-steam and
# steam ships give way through ztr_gfx_modern_ship_trigger (overrides of 03_naval_vehicles.txt, 03_sailingships.txt).
# One to three ships per convoy with the steamers sales, like vanilla.

ztr_motor_cargo_ships = {
	valid_spline_styles = { "naval_route" "coastal_naval_route" }

	speed = 1.8
	min_caravan_distance = 35.0
	max_count = 50
	max_path_length = 10
	side_offset = 1
	path_generation_mode = naval

	possible = {
		ztr_gfx_motor_ship_trigger = yes
		scope:state = {
			market = {
				mg:steamers = {
					market_goods_sell_orders > 10
				}
			}
		}
	}

	route_entities = {
		motor_cargo_ship = {
			entity = "ztr_eu_ship_cargo_1940s_01_entity"
			position = 0
			entity_length = 7.4

			count = {
				value = 0
				scope:state = {
					market = {
						mg:steamers = {
							add = market_goods_sell_orders
						}
					}
				}
				divide = 4000
				add = 1
				max = 3
			}
		}
	}
}
