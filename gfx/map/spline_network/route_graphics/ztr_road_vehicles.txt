# Lorries of the post-1936 road PMs of the railway building (models: scripts/city_gen/ztr_vehiclegen.py).
# 1960s truck from highways on, modern tractor and semi-trailer added with expressways: two types with their own
# speed, both driving afterwards. The vanilla flatbed truck stays off these states (gfx_truck_condition_trigger).
# Tractor and trailer are separate entities on equal slots (entity scale 1.5: mesh slot x 1.5), so the trailer
# follows the curves; the trailer mesh reaches forward over the tractor's coupling. Vans and light trucks share
# the modern trigger in a third type; buses come with the urban centre transport PMs.

ztr_truck_1960s_route_graphics = {
	valid_spline_styles = { "dirtroad" "pavedroad" }

	speed = 0.6
	min_caravan_distance = 10.0
	max_count = 200
	max_path_length = 4

	possible = {
		ztr_gfx_truck_1960s_trigger = yes
	}

	route_entities = {
		truck = {
			entity = "ztr_eu_truck_1960s_truck_01_entity"
			position = 0
			entity_length = 1.65
		}
		trailer = {
			entity = "ztr_eu_truck_1960s_trailer_01_entity"
			position = 1
			entity_length = 1.65
			# Drawbar trailer on about half of the trucks
			count = {
				integer_range = {
					min = 0
					max = 1
				}
			}
		}
	}
}
